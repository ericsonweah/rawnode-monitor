#!/usr/bin/env node
'use strict';

import { readdir } from 'node:fs/promises';
import path from 'node:path';
import { spawn } from 'node:child_process';

async function walk(root, out = []) {
  for (const entry of await readdir(root, { withFileTypes: true })) {
    if (entry.name === 'node_modules' || entry.name === '.git') continue;
    const absolute = path.join(root, entry.name);
    if (entry.isDirectory()) await walk(absolute, out);
    else if (entry.isFile() && entry.name.endsWith('.js')) out.push(absolute);
  }
  return out;
}

function check(file) {
  return new Promise((resolve) => {
    const child = spawn(process.execPath, ['--check', file], { stdio: ['ignore', 'ignore', 'pipe'] });
    let stderr = '';
    child.stderr.on('data', (chunk) => { stderr += chunk; });
    child.on('close', (code) => resolve({ file, code, stderr }));
  });
}

const files = await walk(process.cwd());
const failures = [];
let cursor = 0;
await Promise.all(Array.from({ length: Math.min(8, files.length || 1) }, async () => {
  while (cursor < files.length) {
    const result = await check(files[cursor++]);
    if (result.code !== 0) failures.push(result);
  }
}));
if (failures.length) {
  for (const failure of failures) process.stderr.write(`${failure.file}\n${failure.stderr}\n`);
  process.exitCode = 1;
} else process.stdout.write(`Checked ${files.length} JavaScript files: OK\n`);

'use strict';

const bool = (name, fallback = false) => {
  const value = process.env[name];
  if (value == null || value === '') return fallback;
  return ['1', 'true', 'yes', 'on', 'enabled'].includes(String(value).toLowerCase());
};

const int = (name, fallback) => {
  const value = Number(process.env[name]);
  return Number.isFinite(value) ? Math.trunc(value) : fallback;
};

export default {
  stateDir: process.env.RAWNODE_PM_HOME || undefined,

  runtime: {
    enabled: bool('RAWNODE_PM_RUNTIME', process.env.RAWNODE_PM_MANAGED === '1'),
    autoReadyAfterBoot: bool('RAWNODE_PM_AUTO_READY', false),
    captureSignals: process.env.RAWNODE_PM_CAPTURE_SIGNALS !== 'false',
    telemetryIntervalMs: int('RAWNODE_PM_TELEMETRY_INTERVAL', 2_000),
    incidentDir: process.env.RAWNODE_PM_INCIDENT_DIR || 'storage/framework/pm/incidents',
  },
};

#!/usr/bin/env node
'use strict';

import { access, readFile } from 'node:fs/promises';
import { spawn } from 'node:child_process';
import path from 'node:path';

const root = process.cwd();
const pkg = JSON.parse(await readFile(path.join(root, 'package.json'), 'utf8'));
const declarations = new Set();

for (const entry of Object.values(pkg.exports || {})) {
  if (entry && typeof entry === 'object' && typeof entry.types === 'string') declarations.add(entry.types);
}
if (typeof pkg.types === 'string') declarations.add(pkg.types);
if (!declarations.size) throw new Error('package.json must publish at least one declaration file.');

for (const relative of declarations) {
  await access(path.join(root, relative.replace(/^\.\//, '')));
}

const source = await readFile(path.join(root, 'types/index.d.ts'), 'utf8');
for (const required of [
  'ProcessDefinition',
  'ProcessManager',
  'Supervisor',
  'ControlClient',
  'RuntimeAgent',
  'RawNodeRuntimeAdapter',
  'StartupGenerator',
]) {
  if (!new RegExp(`export\\s+(?:declare\\s+)?class\\s+${required}\\b`).test(source)) {
    throw new Error(`Missing public declaration for ${required}.`);
  }
}

const result = await new Promise((resolve) => {
  const child = spawn('tsc', ['-p', 'tsconfig.types.json'], {
    cwd: root,
    stdio: 'inherit',
    windowsHide: true,
  });
  child.once('error', (error) => resolve(error?.code === 'ENOENT' ? { skipped: true } : { error }));
  child.once('exit', (code) => resolve({ code }));
});

if (result.error) throw result.error;
if (result.code != null && result.code !== 0) process.exitCode = result.code;
else process.stdout.write(`Checked ${declarations.size} declaration target(s)${result.skipped ? ' (TypeScript compiler unavailable; structural checks only)' : ': OK'}\n`);

  const stateDir = path.join(root, 'state');
  const paths = resolveStatePaths(stateDir);
  const occupied = http.createServer((_req, res) => res.end('occupied'));
  await new Promise((resolve, reject) => {
    occupied.once('error', reject);
    occupied.listen(0, '127.0.0.1', resolve);
  });
  const port = occupied.address().port;

  const failed = new ProcessManager({
    stateDir,
    cwd: root,
    managementEnabled: true,
    managementHost: '127.0.0.1',
    managementPort: port,
  });

  try {
    await assert.rejects(
      () => failed.start(),
      (error) => error?.code === 'EADDRINUSE',
    );
    assert.equal(failed.closed, true);
    await assert.rejects(() => stat(paths.daemonLock), (error) => error?.code === 'ENOENT');
    await assert.rejects(() => stat(paths.daemonPid), (error) => error?.code === 'ENOENT');
    if (process.platform !== 'win32') {
      await assert.rejects(() => stat(paths.socket), (error) => error?.code === 'ENOENT');
    }
  } finally {
    await new Promise((resolve) => occupied.close(resolve));
  }

  const recovered = new ProcessManager({ stateDir, cwd: root, managementEnabled: false });
  await recovered.start();
  try {
    assert.equal(recovered.health().ok, true);
  } finally {
    await recovered.close();

test('supervisor starts, observes, diagnoses, and gracefully stops a fork worker', async () => {
  const root = await mkdtemp(path.join(os.tmpdir(), 'rawnode-pm-supervisor-'));
  const registry = new ProcessRegistry();
  const logs = new LogMultiplexer({ root: path.join(root, 'logs') });
  const metrics = new MetricsAggregator({ historyLimit: 10 });
  const supervisor = new Supervisor({ registry, logs, metrics, healthTickMs: 100 });
  try {
    const status = await supervisor.start({
      name: 'api',
      script: fixture,
      mode: 'fork',
      env: { READY_DELAY_MS: '10', RAWNODE_PM_TELEMETRY_INTERVAL: '50' },
      readiness: { mode: 'ipc', timeoutMs: 3_000 },
      health: { intervalMs: 100, timeoutMs: 500, heartbeatTimeoutMs: 1_000 },
      telemetry: { intervalMs: 50 },
      logs: { bufferBytes: 128 * 1024 },
    });
    assert.equal(status.activeInstances, 1);
    assert.equal(status.processes[0].state, 'running');
    await sleep(250);
    const running = registry.active('api', 0);
    assert.ok(running.metrics?.memory?.rss > 0);
    assert.ok(logs.query('api', { lines: 20 }).some((entry) => entry.line.includes('online:')));
    const diagnostics = await supervisor.diagnostics('api', { timeoutMs: 2_000 });
    assert.equal(diagnostics.diagnostics[0].result.fixture, true);

    const stopped = await supervisor.stop('api');
    assert.equal(stopped.activeInstances, 0);
    assert.ok(stopped.processes.some((entry) => entry.state === 'stopped'));
    assert.ok(logs.query('api', { lines: 50 }).some((entry) => entry.line.includes('draining:')));
  } finally {
    await supervisor.close();
    await logs.close();
  }
});

  const registry = new ProcessRegistry();
  const supervisor = new Supervisor({ registry, healthTickMs: 100 });

});

'use strict';

import net from 'node:net';
import { sleep, withTimeout } from '../support/async.js';

async function httpProbe(probe) {
  const url = probe.url || `http://${probe.host}:${probe.port}${probe.path || '/'}`;
  const response = await fetch(url, { method: 'GET', redirect: 'manual', signal: AbortSignal.timeout(probe.timeoutMs) });
  return response.status === probe.expectedStatus;
}

async function tcpProbe(probe) {
  return withTimeout(new Promise((resolve, reject) => {
    const socket = net.connect({ host: probe.host, port: probe.port });
    socket.once('connect', () => { socket.destroy(); resolve(true); });
    socket.once('error', reject);
  }), probe.timeoutMs, { message: `TCP readiness probe timed out after ${probe.timeoutMs}ms` });
}

export async function waitForProbe(probe, options = {}) {
  const deadline = Date.now() + probe.timeoutMs;
  let lastError = null;
  while (Date.now() < deadline) {
    if (options.signal?.aborted) throw options.signal.reason || new Error('Probe aborted');
    try {
      const ok = probe.mode === 'http' ? await httpProbe(probe) : await tcpProbe(probe);
      if (ok) return true;
    } catch (error) { lastError = error; }
    await sleep(probe.intervalMs, options.signal);
  }
  const error = new Error(`Readiness probe '${probe.mode}' failed for ${probe.url || `${probe.host}:${probe.port}`}.`);
  error.code = 'RNPM_READINESS_FAILED';
  error.cause = lastError;
  throw error;
}

'use strict';

import { EventEmitter } from 'node:events';

export class DriverHandle extends EventEmitter {
  constructor(options = {}) {
    super();
    this.kind = options.kind || 'unknown';
    this.child = options.child || null;
    this.worker = options.worker || null;
    this.pid = options.pid || this.child?.pid || this.worker?.process?.pid || null;
    this.stdout = options.stdout || this.child?.stdout || this.worker?.process?.stdout || null;
    this.stderr = options.stderr || this.child?.stderr || this.worker?.process?.stderr || null;
    this.connected = options.connected !== false;
    this._send = options.send || null;
    this._kill = options.kill || null;
    this._disconnect = options.disconnect || null;
  }

  async send(message) {
    if (!this._send) return false;
    return this._send(message);
  }

  kill(signal = 'SIGTERM') {
    if (this._kill) return this._kill(signal);
    return false;
  }

  disconnect() {
    this.connected = false;
    if (this._disconnect) return this._disconnect();
    return false;
  }
}

export default DriverHandle;

function sendAsync(worker, message) {

    if (!cluster.isPrimary) throw new Error('ClusterDriver must run in the supervisor primary process.');
    this.options = options;
    // Node cluster settings are process-global. Serializing setupPrimary()+fork()
    // prevents definitions for concurrent applications from crossing over.
    this._spawnLock = new AsyncLock();
  }

  async spawn(definition, context = {}) {
    return this._spawnLock.run(async () => {
      cluster.setupPrimary({
        exec: definition.script,
        args: [...definition.args],
        execArgv: [...definition.nodeArgs],
        cwd: definition.cwd,
        silent: true,
        serialization: 'advanced',
        windowsHide: true,
        uid: definition.uid,
        gid: definition.gid,
      });

        ...definition.env,
        RAWNODE_PM_MANAGED: '1',
        RAWNODE_PM_CLUSTER: '1',
        RAWNODE_PM_APP: definition.name,
        RAWNODE_PM_SLOT: String(context.slot ?? 0),
        RAWNODE_PM_GENERATION: String(context.generation ?? 0),
        RAWNODE_PM_TELEMETRY_INTERVAL: String(definition.telemetry.intervalMs),
      });
      const handle = new DriverHandle({
        kind: 'cluster',
        worker,
        child: worker.process,
        send: (message) => sendAsync(worker, message),
        kill: (signal) => worker.process.kill(signal),
        disconnect: () => worker.disconnect(),

'use strict';

import os from 'node:os';
import path from 'node:path';
import { ConfigurationError } from '../errors.js';
import { durationMs } from '../support/duration.js';

const VALID_NAME = /^[a-zA-Z0-9][a-zA-Z0-9._-]{0,127}$/;
const VALID_MODES = new Set(['fork', 'cluster', 'external']);
const VALID_READINESS = new Set(['ipc', 'immediate', 'http', 'tcp']);
const VALID_WATCH_MODE = new Set(['reload', 'restart', 'none']);

function bool(value, fallback = false) {
  if (value == null) return fallback;
  if (typeof value === 'boolean') return value;
  return ['1', 'true', 'yes', 'on', 'enabled'].includes(String(value).toLowerCase());
}

function int(value, fallback, min = 0, max = Number.MAX_SAFE_INTEGER) {
  const number = Number(value);
  if (!Number.isFinite(number)) return fallback;
  const out = Math.trunc(number);
  return out >= min && out <= max ? out : fallback;
}

function stringArray(value) {
  if (value == null || value === false) return [];
  if (Array.isArray(value)) return value.map(String).filter(Boolean);
  return String(value).split(',').map((item) => item.trim()).filter(Boolean);
}

function resolveInstances(value, mode) {
  if (mode !== 'cluster') return 1;
  if (value === 'max' || value === 'auto' || value === 0) return Math.max(1, os.availableParallelism?.() || os.cpus().length || 1);
  return int(value, 1, 1, 1024);
}

function normalizeProbe(value, defaults = {}) {
  const input = value && typeof value === 'object' ? value : {};
  const mode = String(input.mode || defaults.mode || 'ipc').toLowerCase();
  if (!VALID_READINESS.has(mode)) throw new ConfigurationError(`Invalid readiness mode '${mode}'.`);
  return Object.freeze({
    mode,
    timeoutMs: durationMs(input.timeoutMs ?? input.timeout ?? defaults.timeoutMs, 30_000),
    intervalMs: durationMs(input.intervalMs ?? defaults.intervalMs, 250),
    url: input.url || defaults.url || null,
    host: input.host || defaults.host || '127.0.0.1',
    port: input.port == null ? defaults.port ?? null : int(input.port, null, 1, 65535),
    path: input.path || defaults.path || '/health/ready',
    expectedStatus: int(input.expectedStatus ?? defaults.expectedStatus, 200, 100, 599),
  });
}

export class ProcessDefinition {
  constructor(input = {}, options = {}) {
    const cwd = path.resolve(input.cwd || options.cwd || process.cwd());
    const name = String(input.name || '').trim();
    if (!VALID_NAME.test(name)) throw new ConfigurationError(`Invalid process name '${name}'. Use letters, numbers, dots, dashes, or underscores.`);

    const mode = String(input.mode || input.execMode || 'fork').toLowerCase();
    if (!VALID_MODES.has(mode)) throw new ConfigurationError(`Invalid process mode '${mode}'.`);

    const scriptValue = input.script || input.entry || input.command;
    if (!scriptValue) throw new ConfigurationError(`Process '${name}' requires script or command.`);
    const script = mode === 'external' ? String(scriptValue) : path.resolve(cwd, String(scriptValue));

    const watchInput = input.watch && typeof input.watch === 'object' ? input.watch : { enabled: input.watch === true };
    const watchMode = String(watchInput.mode || (mode === 'cluster' ? 'reload' : 'restart')).toLowerCase();
    if (!VALID_WATCH_MODE.has(watchMode)) throw new ConfigurationError(`Invalid watch mode '${watchMode}'.`);

    this.name = name;
    this.namespace = String(input.namespace || 'default');
    this.cwd = cwd;
    this.script = script;
    this.args = Object.freeze(stringArray(input.args));
    this.nodeArgs = Object.freeze(stringArray(input.nodeArgs || input.execArgv));
    this.mode = mode;
    this.instances = resolveInstances(input.instances ?? input.workers, mode);
    this.interpreter = input.interpreter || (mode === 'external' ? null : process.execPath);
    this.envOverrides = Object.freeze({ ...(input.env || {}) });
    this.env = Object.freeze({ ...process.env, ...this.envOverrides });
    this.autostart = input.autostart !== false;
    this.autorestart = input.autorestart !== false;
    this.readiness = normalizeProbe(input.readiness, { mode: mode === 'external' ? 'immediate' : 'ipc' });
    this.health = Object.freeze({
      enabled: input.health?.enabled !== false,
      intervalMs: durationMs(input.health?.intervalMs, 5_000),
      timeoutMs: durationMs(input.health?.timeoutMs, 2_000),
      heartbeatTimeoutMs: durationMs(input.health?.heartbeatTimeoutMs, 12_000),
      unhealthyThreshold: int(input.health?.unhealthyThreshold, 3, 1, 100),
      restart: input.health?.restart !== false,
    });
    this.restart = Object.freeze({
      enabled: input.restart?.enabled ?? this.autorestart,
      onCleanExit: bool(input.restart?.onCleanExit, false),
      maxRestarts: int(input.restart?.maxRestarts, 10, 0, 100_000),
      windowMs: durationMs(input.restart?.windowMs, 60_000),
      stableMs: durationMs(input.restart?.stableMs, 30_000),
      minDelayMs: durationMs(input.restart?.minDelayMs, 100),
      maxDelayMs: durationMs(input.restart?.maxDelayMs, 30_000),
      factor: Math.max(1, Number(input.restart?.factor) || 2),
      jitter: Math.max(0, Math.min(1, Number(input.restart?.jitter) || 0.2)),
    });
    this.shutdown = Object.freeze({
      graceMs: durationMs(input.shutdown?.graceMs, 15_000),
      killMs: durationMs(input.shutdown?.killMs, 5_000),
      signal: String(input.shutdown?.signal || 'SIGTERM'),
      killSignal: String(input.shutdown?.killSignal || 'SIGKILL'),
    });
    this.watch = Object.freeze({
      enabled: watchInput.enabled === true,
      mode: watchMode,
      paths: Object.freeze(stringArray(watchInput.paths?.length ? watchInput.paths : ['app', 'src', 'config', 'resources'])),
      ignore: Object.freeze(stringArray(watchInput.ignore?.length ? watchInput.ignore : ['node_modules/**', '.git/**', 'storage/**', 'logs/**', '*.log'])),
      debounceMs: durationMs(watchInput.debounceMs, 250),
      pollingIntervalMs: durationMs(watchInput.pollingIntervalMs, 2_000),
      reconcileIntervalMs: durationMs(watchInput.reconcileIntervalMs, 10_000),
    });
    this.logs = Object.freeze({
      enabled: input.logs?.enabled !== false,
      maxBytes: int(input.logs?.maxBytes, 50 * 1024 * 1024, 64 * 1024),
      maxFiles: int(input.logs?.maxFiles, 5, 1, 100),
      bufferBytes: int(input.logs?.bufferBytes, 2 * 1024 * 1024, 64 * 1024),
      timestamp: input.logs?.timestamp !== false,
    });
    this.telemetry = Object.freeze({
      enabled: input.telemetry?.enabled !== false,
      intervalMs: durationMs(input.telemetry?.intervalMs, 2_000),
    });
    this.metadata = Object.freeze({ ...(input.metadata || {}) });
    this.uid = Number.isInteger(input.uid) ? input.uid : undefined;
    this.gid = Number.isInteger(input.gid) ? input.gid : undefined;
    Object.freeze(this);
  }

  toJSON({ includeEnv = true } = {}) {
    return {
      name: this.name,
      namespace: this.namespace,
      cwd: this.cwd,
      script: this.script,
      args: [...this.args],
      nodeArgs: [...this.nodeArgs],
      mode: this.mode,
      instances: this.instances,
      interpreter: this.interpreter,
      ...(includeEnv ? { env: { ...this.envOverrides } } : {}),
      autostart: this.autostart,
      autorestart: this.autorestart,
      readiness: { ...this.readiness },
      health: { ...this.health },
      restart: { ...this.restart },
      shutdown: { ...this.shutdown },
      watch: { ...this.watch, paths: [...this.watch.paths], ignore: [...this.watch.ignore] },
      logs: { ...this.logs },
      telemetry: { ...this.telemetry },
      metadata: { ...this.metadata },
      uid: this.uid,
      gid: this.gid,
    };
  }

  static from(input, options) {
    return input instanceof ProcessDefinition ? input : new ProcessDefinition(input, options);
  }
}

export default ProcessDefinition;

'use strict';

import path from 'node:path';
import { access } from 'node:fs/promises';
import { pathToFileURL } from 'node:url';
import { ProcessDefinition } from './ProcessDefinition.js';
import { ConfigurationError } from '../errors.js';

const CANDIDATES = Object.freeze([
  'rawnode.process.mjs',
  'rawnode.process.js',
  'rawnode-pm.config.mjs',
  'rawnode-pm.config.js',
  'config/process.js',
]);

async function firstExisting(cwd, values) {
  for (const value of values) {
    const absolute = path.resolve(cwd, value);
    try { await access(absolute); return absolute; } catch {}
  }
  return null;
}

function normalizeManifest(value) {
  if (Array.isArray(value)) return { apps: value };
  if (value && Array.isArray(value.apps)) return value;
  if (value && Array.isArray(value.processes)) return { ...value, apps: value.processes };
  if (value && value.name && (value.script || value.command || value.entry)) return { apps: [value] };
  throw new ConfigurationError('Process manifest must export an app definition, an array, or { apps: [] }.');
}

export class ManifestLoader {
  constructor(options = {}) {
    this.cwd = path.resolve(options.cwd || process.cwd());
  }

  async resolve(file = null) {
    if (file) return path.resolve(this.cwd, file);
    const found = await firstExisting(this.cwd, CANDIDATES);
    if (!found) throw new ConfigurationError(`No process manifest found. Tried: ${CANDIDATES.join(', ')}`);
    return found;
  }

  async load(file = null, options = {}) {
    const absolute = await this.resolve(file);
    const url = pathToFileURL(absolute);
    if (options.fresh !== false) url.searchParams.set('t', String(Date.now()));
    const mod = await import(url.href);
    const exported = mod.default ?? mod.processes ?? mod.apps ?? mod;
    const manifest = normalizeManifest(typeof exported === 'function' ? await exported({ cwd: this.cwd, env: process.env }) : exported);
    const apps = manifest.apps.map((definition) => ProcessDefinition.from(definition, { cwd: path.dirname(absolute) }));
    const names = new Set();
    for (const app of apps) {
      if (names.has(app.name)) throw new ConfigurationError(`Duplicate process name '${app.name}' in manifest.`);
      names.add(app.name);
    }
    return { file: absolute, options: { ...(manifest.options || {}) }, apps };
  }
}

'use strict';

const SECRET_KEY = /(?:secret|token|password|passwd|private[_-]?key|api[_-]?key|credential|authorization|cookie|session[_-]?key|pepper|dsn)/i;
const BEARER = /\bBearer\s+[A-Za-z0-9._~+\/-]{1,8192}=*/gi;
// Bounded components avoid regex backtracking becoming quadratic on very long log lines.
const URI_CREDENTIALS = /\b([a-z][a-z0-9+.-]{0,31}:\/\/)([^:@/\s]{1,256}):([^@/\s]{1,256})@/gi;

export class SecretRedactor {
  constructor(options = {}) {
    this.placeholder = options.placeholder || '[redacted]';
    this.additionalKeys = new Set((options.keys || []).map((key) => String(key).toLowerCase()));
    this.maxDepth = Math.max(1, Number(options.maxDepth) || 16);
    this.maxArray = Math.max(1, Number(options.maxArray) || 1_000);
  }

  isSecretKey(key) {
    const value = String(key || '');
    return SECRET_KEY.test(value) || this.additionalKeys.has(value.toLowerCase());
  }

  text(value) {
    return String(value ?? '')
      .replace(BEARER, `Bearer ${this.placeholder}`)
      .replace(URI_CREDENTIALS, `$1${this.placeholder}:${this.placeholder}@`);
  }

  redact(value, key = '', depth = 0, seen = new WeakMap()) {
    if (this.isSecretKey(key) && value != null) return this.placeholder;
    if (typeof value === 'string') return this.text(value);
    if (typeof value === 'bigint') return `${value}n`;
    if (value == null || typeof value !== 'object') return value;
    if (depth >= this.maxDepth) return '[max-depth]';
    if (seen.has(value)) return '[circular]';
    seen.set(value, true);

    if (Buffer.isBuffer(value)) return `[buffer:${value.length}]`;
    if (value instanceof Date) return value.toISOString();
    if (value instanceof Error) {
      return {
        name: value.name,
        code: value.code || null,
        message: this.text(value.message),
      };
    }
    if (Array.isArray(value)) {
      return value.slice(0, this.maxArray).map((item) => this.redact(item, '', depth + 1, seen));
    }

    const out = Object.create(null);
    for (const [childKey, childValue] of Object.entries(value)) {
      out[childKey] = this.redact(childValue, childKey, depth + 1, seen);
    }
    return out;
  }
}

export const defaultRedactor = new SecretRedactor();
export default SecretRedactor;

import { randomBytes, timingSafeEqual } from 'node:crypto';

function safeEqual(a, b) {
  const left = Buffer.from(String(a || ''), 'utf8');
  const right = Buffer.from(String(b || ''), 'utf8');
  if (left.length !== right.length || left.length === 0) return false;
  return timingSafeEqual(left, right);
}

export class ControlAuthenticator {
  constructor(options = {}) {
    this.tokenPath = options.tokenPath;
    this.token = options.token || null;
  }

  async initialize() {
    if (this.token) return this.token;
    if (!this.tokenPath) throw new Error('ControlAuthenticator requires tokenPath or token.');
    await ensureDir(path.dirname(this.tokenPath), 0o700);
    try {
      const existing = (await readFile(this.tokenPath, 'utf8')).trim();
      if (existing.length >= 32) {
        this.token = existing;
        return existing;
      }
    } catch (error) {
      if (error?.code !== 'ENOENT') throw error;
    }
    const token = randomBytes(32).toString('hex');
    await writePrivateFile(this.tokenPath, `${token}\n`);
    this.token = token;
    return token;
  }

  verify(candidate) {
    return safeEqual(this.token, candidate);
  }

  keyBytes() {
    const token = String(this.token || '');
    if (/^[a-f0-9]{64}$/i.test(token)) return Buffer.from(token, 'hex');
    return Buffer.from(token.padEnd(32, '0').slice(0, 32), 'utf8');
  }
}

'use strict';

import { EventEmitter } from 'node:events';
import path from 'node:path';
import { mkdir } from 'node:fs/promises';
import { withTimeout } from '../support/async.js';
import { serializeError } from '../errors.js';

function safeResolve(app, key, fallback = null) {
  try {
    if (app?.container?.has?.(key) || typeof app?.resolve === 'function') {
      const value = app.resolve(key);
      return value === undefined ? fallback : value;
    }
  } catch {}
  return fallback;
}

function callSnapshot(value) {
  if (!value) return null;
  for (const method of ['snapshot', 'stats', 'getStats', 'metrics', 'health']) {
    try {
      if (typeof value[method] === 'function') return value[method]();
    } catch (error) {
      return { ok: false, error: error?.message || String(error) };
    }
  }
  return null;
}

function underlyingServer(server) {
  return server?.httpServer || server?.server || server?.raw || server;
}

export class RawNodeRuntimeAdapter extends EventEmitter {
  constructor(options = {}) {
    super();
    this.app = options.app || null;
    this.logger = options.logger || safeResolve(this.app, 'logger', console) || console;
    this.incidentDir = path.resolve(options.incidentDir || process.env.RAWNODE_PM_INCIDENT_DIR || 'storage/framework/pm/incidents');
    this.servers = new Map();
    this.resources = new Map();
    this.activeRequests = 0;
    this.totalRequests = 0;
    this.sockets = new Set();
    this._draining = false;
    for (const entry of options.servers || []) this.trackServer(entry);
  }

  trackServer(server, options = {}) {
    if (!server) return this;
    const name = options.name || `server-${this.servers.size + 1}`;
    const raw = underlyingServer(server);
    const record = { name, server, raw, sockets: new Set(), close: options.close || null, forceClose: options.forceClose || null };
    this.servers.set(name, record);

    if (raw && typeof raw.on === 'function') {
      raw.on('connection', (socket) => {
        record.sockets.add(socket);
        this.sockets.add(socket);
        socket.once('close', () => {
          record.sockets.delete(socket);
          this.sockets.delete(socket);
        });
      });
      raw.on('request', (_req, res) => {
        this.activeRequests++;
        this.totalRequests++;
        let finished = false;
        const done = () => {
          if (finished) return;
          finished = true;
          this.activeRequests = Math.max(0, this.activeRequests - 1);
        };
        res.once?.('finish', done);
        res.once?.('close', done);
      });
    }
    return this;
  }

  registerResource(name, resource) {
    if (!name || !resource) throw new Error('registerResource(name, resource) requires both values.');
    this.resources.set(String(name), resource);
    return this;
  }

  async health({ deep = true } = {}) {
    const health = safeResolve(this.app, 'health') || safeResolve(this.app, 'http.health') || safeResolve(this.app, 'health.check');
    let framework = null;
    try {
      if (health?.snapshot) framework = await health.snapshot({ deep });
      else framework = await Promise.resolve(callSnapshot(health));
    } catch (error) {
      framework = { ok: false, ready: false, error: error?.message || String(error) };
    }

    const resources = Object.create(null);
    let resourcesOk = true;
    for (const [name, resource] of this.resources) {
      if (typeof resource.health !== 'function') continue;
      try {
        const value = await resource.health();
        resources[name] = value ?? { ok: true };
        if (value?.ok === false) resourcesOk = false;
      } catch (error) {
        resources[name] = { ok: false, error: error?.message || String(error) };
        resourcesOk = false;
      }
    }

    return {
      ok: framework?.ok !== false && resourcesOk,
      ready: framework?.ready !== false && !this._draining,
      draining: this._draining,
      activeRequests: this.activeRequests,
      openSockets: this.sockets.size,
      framework,
      resources,
    };
  }

  async metrics() {
    const sources = {
      http: safeResolve(this.app, 'http.metrics'),
      events: safeResolve(this.app, 'events'),
      cache: safeResolve(this.app, 'cache'),
      redis: safeResolve(this.app, 'redis'),
      session: safeResolve(this.app, 'session.metrics'),
      websocket: safeResolve(this.app, 'ws') || safeResolve(this.app, 'io'),
      tcp: safeResolve(this.app, 'tcp'),
      udp: safeResolve(this.app, 'udp'),
      mongo: safeResolve(this.app, 'mongo'),
      view: safeResolve(this.app, 'view.observability') || safeResolve(this.app, 'view.cache'),
    };
    const framework = Object.create(null);
    for (const [name, source] of Object.entries(sources)) {
      try { framework[name] = await Promise.resolve(callSnapshot(source)); }
      catch (error) { framework[name] = { error: error?.message || String(error) }; }
    }
    for (const [name, resource] of this.resources) {
      if (typeof resource.metrics !== 'function') continue;
      try { framework[name] = await resource.metrics(); }
      catch (error) { framework[name] = { error: error?.message || String(error) }; }
    }
    return {
      activeRequests: this.activeRequests,
      totalRequests: this.totalRequests,
      openSockets: this.sockets.size,
      framework,
    };
  }

  async drain(options = {}) {
    if (this._draining) return { alreadyDraining: true };
    this._draining = true;
    const deadlineMs = Math.max(1, Number(options.deadlineMs) || 15_000);
    const events = safeResolve(this.app, 'events');
    try { await events?.dispatch?.('pm:runtime:draining', { pid: process.pid, deadlineMs }); } catch {}

    const resourceResults = Object.create(null);
    for (const [name, resource] of this.resources) {
      if (typeof resource.drain !== 'function') continue;
      try { resourceResults[name] = await withTimeout(() => resource.drain(options), deadlineMs); }
      catch (error) { resourceResults[name] = { ok: false, error: error?.message || String(error) }; }
    }

    const closes = [];
    for (const record of this.servers.values()) {
      closes.push(this._closeServer(record, deadlineMs));
    }
    const servers = await Promise.allSettled(closes);
    return {
      ok: servers.every((result) => result.status === 'fulfilled'),
      activeRequests: this.activeRequests,
      openSockets: this.sockets.size,
      resources: resourceResults,
      servers: servers.map((result) => result.status === 'fulfilled' ? result.value : { ok: false, error: result.reason?.message || String(result.reason) }),
    };
  }

  async _closeServer(record, deadlineMs) {
    const close = record.close || record.server?.close?.bind(record.server) || record.raw?.close?.bind(record.raw);
    if (typeof close !== 'function') return { name: record.name, ok: true, skipped: true };
    const task = new Promise((resolve, reject) => {
      let settled = false;
      const done = (error) => {
        if (settled) return;
        settled = true;
        error ? reject(error) : resolve({ name: record.name, ok: true });
      };
      try {
        const value = close(done);
        if (value && typeof value.then === 'function') value.then(() => done(), done);
        else if (close.length === 0) done();
      } catch (error) { done(error); }
    });
    try {
      return await withTimeout(task, deadlineMs, { message: `Server '${record.name}' did not drain within ${deadlineMs}ms.` });
    } catch (error) {
      if (typeof record.forceClose === 'function') {
        try { await record.forceClose(); } catch {}
      }
      for (const socket of record.sockets) {
        try { socket.destroy(); } catch {}
      }
      return { name: record.name, ok: false, forced: true, error: error?.message || String(error) };
    }
  }

  async shutdown(options = {}) {
    const events = safeResolve(this.app, 'events');
    try { await events?.dispatch?.('pm:runtime:shutting-down', { pid: process.pid }); } catch {}
    for (const [name, resource] of Array.from(this.resources).reverse()) {
      if (typeof resource.shutdown !== 'function') continue;
      try { await resource.shutdown(options); }
      catch (error) { this.logger?.warn?.(`[rawnode-pm] Resource '${name}' shutdown failed`, error); }
    }
    if (this.app?.shutdown) await this.app.shutdown();
    return { ok: true };
  }

  async reloadConfig() {
    const config = safeResolve(this.app, 'config');
    if (config?.reload) return config.reload();
    return { ok: false, skipped: true, reason: 'config.reload unavailable' };
  }

  async diagnostics(options = {}) {
    if (!process.report?.writeReport) return { ok: false, supported: false };
    await mkdir(this.incidentDir, { recursive: true, mode: 0o700 });
    const file = path.join(this.incidentDir, `report-${process.pid}-${Date.now()}.json`);
    try {
      const output = process.report.writeReport(file, options.error);
      return { ok: true, file: output || file };
    } catch (error) {
      return { ok: false, error: serializeError(error) };
    }
  }
}

export default RawNodeRuntimeAdapter;

function activeResources() {
  try {
    const list = process.getActiveResourcesInfo?.() || [];
    const counts = Object.create(null);
    for (const item of list) counts[item] = (counts[item] || 0) + 1;
    return counts;
  } catch { return {}; }
}

export class RuntimeAgent extends EventEmitter {
  constructor(options = {}) {
    super();
    this.adapter = options.adapter || new RawNodeRuntimeAdapter(options);
    this.intervalMs = Math.max(250, Number(options.intervalMs || process.env.RAWNODE_PM_TELEMETRY_INTERVAL) || 2_000);
    this.managed = options.managed ?? process.env.RAWNODE_PM_MANAGED === '1';
    this.autoExit = options.autoExit !== false;
    this.captureSignals = options.captureSignals !== false;
    this.started = false;
    this.readyState = false;
    this.draining = false;
    this.stopping = false;
    this._timer = null;
    this._messageHandler = (message) => this._onMessage(message);
    this._sigtermHandler = () => this._onSignal('SIGTERM');
    this._sigintHandler = () => this._onSignal('SIGINT');
    this._exceptionMonitor = (error, origin) => {
      this.send(RUNTIME_MESSAGE.ERROR, { fatal: true, origin, error: serializeError(error, { stack: true }) });
    };
    this._cpu = process.cpuUsage();
    this._cpuAt = performance.now();
    this._elu = performance.eventLoopUtilization();
    this._delay = monitorEventLoopDelay({ resolution: Math.max(10, Number(options.eventLoopResolution) || 20) });

  }

    this.started = true;

    process.on('uncaughtExceptionMonitor', this._exceptionMonitor);

      rss: metrics.memory.rss,

    const adapter = await this.adapter.metrics();

      at: Date.now(),
      uptimeMs: Math.round(process.uptime() * 1_000),
      memory,
      cpu: {
        userMicros: cpu.user,
        systemMicros: cpu.system,
        percent: ((cpu.user + cpu.system) / elapsedUs) * 100,
      },
      eventLoop: {
        utilization: elu.utilization,
        activeMs: elu.active,
        idleMs: elu.idle,
        delayMinMs: Number.isFinite(this._delay.min) ? this._delay.min / 1e6 : 0,
        delayMeanMs: Number.isFinite(this._delay.mean) ? this._delay.mean / 1e6 : 0,
        delayMaxMs: Number.isFinite(this._delay.max) ? this._delay.max / 1e6 : 0,
        delayP95Ms: this._delay.percentile(95) / 1e6,
        delayP99Ms: this._delay.percentile(99) / 1e6,

    const requestId = message.requestId || null;
    try {
      switch (message.type) {
        case SUPERVISOR_MESSAGE.DRAIN: {
          const result = await this.drain(message.payload || {});
          this.send(RUNTIME_MESSAGE.DRAINED, result, { replyTo: requestId });
          break;
        }
        case SUPERVISOR_MESSAGE.SHUTDOWN: {
          const result = await this.shutdown(message.payload || {});
          this.send(RUNTIME_MESSAGE.SHUTDOWN_COMPLETE, result, { replyTo: requestId });
          if (this.autoExit) setImmediate(() => process.exit(0));
          break;
        }
        case SUPERVISOR_MESSAGE.HEALTH:
          this.send(RUNTIME_MESSAGE.HEALTH, await this.health(), { replyTo: requestId });
          break;
        case SUPERVISOR_MESSAGE.METRICS:
          this.send(RUNTIME_MESSAGE.METRICS, await this.metrics(), { replyTo: requestId });
          break;
        case SUPERVISOR_MESSAGE.DIAGNOSTICS:
          this.send(RUNTIME_MESSAGE.DIAGNOSTIC_READY, await this.adapter.diagnostics(message.payload || {}), { replyTo: requestId });
          break;
        case SUPERVISOR_MESSAGE.RELOAD_CONFIG:
          this.send('runtime:config-reloaded', await this.adapter.reloadConfig(), { replyTo: requestId });
          break;
        default:
          break;
      }
    } catch (error) {
      this.send(RUNTIME_MESSAGE.ERROR, { fatal: false, origin: message.type, error: serializeError(error, { stack: true }) }, { replyTo: requestId });
    }
  }

    this.draining = true;
    this.readyState = false;

  }

    if (this.stopping) return { ok: true, alreadyStopping: true };
    this.stopping = true;
    if (!this.draining) await this.drain(options);
    const result = await this.adapter.shutdown(options);
    await this.close();
    return result;
  }

  async _onSignal(signal) {
    if (this.stopping) return;
    try { await this.shutdown({ signal, deadlineMs: 15_000 }); }
    catch (error) { this.send(RUNTIME_MESSAGE.ERROR, { fatal: false, origin: signal, error: serializeError(error) }); }
    if (this.autoExit) process.exitCode = 0;
  }

  async close() {
    if (this._timer) clearInterval(this._timer);
    this._timer = null;
    try { this._delay.disable(); } catch {}
    process.off('message', this._messageHandler);
    process.off('uncaughtExceptionMonitor', this._exceptionMonitor);
    process.off('SIGTERM', this._sigtermHandler);
    process.off('SIGINT', this._sigintHandler);
    this.started = false;
  }
}

import { RUNTIME_PROTOCOL_VERSION } from '../constants.js';

  return {
    __rawnodePm: 1,
    protocol: RUNTIME_PROTOCOL_VERSION,
    type,
    at: Date.now(),
    pid: process.pid,
    app: process.env.RAWNODE_PM_APP || meta.app || null,
    slot: Number(process.env.RAWNODE_PM_SLOT ?? meta.slot ?? 0),
    generation: Number(process.env.RAWNODE_PM_GENERATION ?? meta.generation ?? 0),
    requestId: meta.requestId || null,
    replyTo: meta.replyTo || null,
    payload,
  };
}

export function isRuntimePacket(value) {
  return !!(value && typeof value === 'object' && value.__rawnodePm === 1 && typeof value.type === 'string');

'use strict';

import { BackoffPolicy } from './BackoffPolicy.js';

export class RestartPolicy {
  constructor(options = {}) {
    this.options = {
      enabled: options.enabled !== false,
      onCleanExit: options.onCleanExit === true,
      maxRestarts: Math.max(0, Number(options.maxRestarts) || 10),
      windowMs: Math.max(1_000, Number(options.windowMs) || 60_000),
      stableMs: Math.max(0, Number(options.stableMs) || 30_000),
    };
    this.backoff = new BackoffPolicy(options);
    this.restartTimes = [];
    this.attempt = 0;
    this.lastStableAt = 0;
  }

  prune(now = Date.now()) {
    const cutoff = now - this.options.windowMs;
    while (this.restartTimes.length && this.restartTimes[0] < cutoff) this.restartTimes.shift();
  }

  markStable(now = Date.now()) {
    this.attempt = 0;
    this.lastStableAt = now;
    this.prune(now);
  }

  noteRuntime(startedAt, now = Date.now()) {
    if (startedAt && now - startedAt >= this.options.stableMs) this.markStable(now);
  }

  decide({ expected = false, exitCode = null, signal = null, startedAt = 0 } = {}) {
    const now = Date.now();
    this.noteRuntime(startedAt, now);
    if (!this.options.enabled || expected) return { restart: false, reason: expected ? 'expected-exit' : 'restart-disabled' };
    if (exitCode === 0 && !signal && !this.options.onCleanExit) return { restart: false, reason: 'clean-exit' };

    this.prune(now);
    if (this.restartTimes.length >= this.options.maxRestarts) {
      return {
        restart: false,
        quarantine: true,
        reason: 'restart-window-exhausted',
        restarts: this.restartTimes.length,
        windowMs: this.options.windowMs,
      };
    }

    this.restartTimes.push(now);
    this.attempt++;
    return {
      restart: true,
      delayMs: this.backoff.delay(this.attempt),
      attempt: this.attempt,
      reason: 'unexpected-exit',
    };
  }

  snapshot() {
    this.prune();
    return {
      options: { ...this.options },
      attempt: this.attempt,
      restartTimes: this.restartTimes.slice(),
      lastStableAt: this.lastStableAt,
    };
  }
}

export default RestartPolicy;

'use strict';

import { EventEmitter } from 'node:events';
import { PROCESS_STATES } from '../constants.js';
import { ProcessLifecycleError } from '../errors.js';

const TRANSITIONS = Object.freeze({
  defined: new Set(['starting', 'stopped']),
  starting: new Set(['online', 'ready', 'running', 'boot_failed', 'crashed', 'stopping']),
  online: new Set(['ready', 'running', 'unhealthy', 'draining', 'stopping', 'crashed']),
  ready: new Set(['running', 'unhealthy', 'draining', 'stopping', 'crashed']),
  running: new Set(['unhealthy', 'draining', 'stopping', 'crashed']),
  unhealthy: new Set(['running', 'draining', 'backoff', 'starting', 'stopping', 'crashed', 'quarantined']),
  draining: new Set(['stopping', 'stopped', 'crashed']),
  stopping: new Set(['stopped', 'crashed']),
  stopped: new Set(['starting']),
  crashed: new Set(['backoff', 'starting', 'quarantined', 'stopped']),
  backoff: new Set(['starting', 'quarantined', 'stopped']),
  boot_failed: new Set(['backoff', 'starting', 'quarantined', 'stopped']),
  quarantined: new Set(['starting', 'stopped']),
});

export class ProcessStateMachine extends EventEmitter {
  constructor(initial = 'defined', options = {}) {
    super();
    if (!PROCESS_STATES.includes(initial)) throw new ProcessLifecycleError(`Unknown process state '${initial}'.`);
    this.state = initial;
    this.changedAt = Date.now();
    this.historyLimit = Math.max(10, Number(options.historyLimit) || 100);
    this.history = [{ from: null, to: initial, at: this.changedAt, reason: 'initial' }];
  }

  can(next) {
    return next === this.state || TRANSITIONS[this.state]?.has(next) === true;
  }

  transition(next, meta = {}) {
    if (!PROCESS_STATES.includes(next)) throw new ProcessLifecycleError(`Unknown process state '${next}'.`);
    if (next === this.state) return this.state;
    if (!this.can(next) && meta.force !== true) {
      throw new ProcessLifecycleError(`Invalid process transition ${this.state} -> ${next}.`, {
        code: 'RNPM_INVALID_TRANSITION',
        details: { from: this.state, to: next, meta },
      });
    }
    const previous = this.state;
    this.state = next;
    this.changedAt = Date.now();
    const entry = { from: previous, to: next, at: this.changedAt, reason: meta.reason || null, details: meta.details || null };
    this.history.push(entry);
    if (this.history.length > this.historyLimit) this.history.splice(0, this.history.length - this.historyLimit);
    this.emit('transition', entry);
    this.emit(next, entry);
    return next;
  }

  snapshot() {
    return { state: this.state, changedAt: this.changedAt, history: this.history.slice() };
  }
}

export default ProcessStateMachine;

'use strict';

export class BackoffPolicy {
  constructor(options = {}) {
    this.minDelayMs = Math.max(0, Number(options.minDelayMs) || 100);
    this.maxDelayMs = Math.max(this.minDelayMs, Number(options.maxDelayMs) || 30_000);
    this.factor = Math.max(1, Number(options.factor) || 2);
    this.jitter = Math.max(0, Math.min(1, Number(options.jitter) || 0.2));
  }

  delay(attempt) {
    const power = Math.max(0, Number(attempt) - 1);
    const raw = Math.min(this.maxDelayMs, this.minDelayMs * (this.factor ** power));
    if (this.jitter <= 0) return Math.round(raw);
    const spread = raw * this.jitter;
    return Math.max(0, Math.round(raw - spread + Math.random() * spread * 2));
  }
}

export default BackoffPolicy;

'use strict';

import { EventEmitter } from 'node:events';
import { randomUUID } from 'node:crypto';
import { ProcessRegistry } from './ProcessRegistry.js';
import { ForkDriver } from '../drivers/ForkDriver.js';
import { ClusterDriver } from '../drivers/ClusterDriver.js';
import { ExternalProcessDriver } from '../drivers/ExternalProcessDriver.js';
import { waitForProbe } from '../drivers/probes.js';
import { Deferred, AsyncLock, sleep, withTimeout, mapLimit } from '../support/async.js';
import { RUNTIME_MESSAGE, SUPERVISOR_MESSAGE } from '../constants.js';
import { runtimePacket, isRuntimePacket } from '../runtime/RuntimeProtocol.js';
import { serializeError, ProcessLifecycleError } from '../errors.js';

function liveState(state) {
  return !['stopped', 'crashed', 'quarantined', 'boot_failed'].includes(state);
}

export class Supervisor extends EventEmitter {
  constructor(options = {}) {
    super();
    this.registry = options.registry || new ProcessRegistry();
    this.logs = options.logs || null;
    this.metrics = options.metrics || null;
    this.journal = options.journal || null;
    this.drivers = {
      fork: options.forkDriver || new ForkDriver(),
      cluster: options.clusterDriver || new ClusterDriver(),
      external: options.externalDriver || new ExternalProcessDriver(),
    };
    this.operationLocks = new Map();
    this.pendingRuntimeRequests = new Map();
    this.closed = false;
    this.healthTickMs = Math.max(500, Number(options.healthTickMs) || 2_000);
    this._healthTimer = setInterval(() => this._healthTick().catch((error) => this.emit('error', error)), this.healthTickMs);
    this._healthTimer.unref?.();
    this.on('error', (error) => options.logger?.error?.('[rawnode-pm] supervisor error', error));
    this._wireRegistry();
  }

  _wireRegistry() {
    this.registry.on('transition', (record, entry) => {
      this.emit('transition', record, entry);
      this.journal?.append?.('process.transition', { id: record.id, name: record.name, slot: record.slot, ...entry }).catch(() => {});
    });
  }

  _lock(name) {
    let lock = this.operationLocks.get(name);
    if (!lock) {
      lock = new AsyncLock();
      this.operationLocks.set(name, lock);
    }
    return lock;
  }

  async start(input, options = {}) {
    const group = typeof input === 'string' ? this.registry.requireGroup(input) : this.registry.define(input);
    return this._lock(group.name).run(() => this._startGroup(group, options));
  }

  async _startGroup(group, options = {}) {
    if (this.closed) throw new ProcessLifecycleError('Supervisor is closed.');
    if (group.quarantined && options.force !== true) {
      throw new ProcessLifecycleError(`Process '${group.name}' is quarantined. Start with force after correcting the failure.`, {
        code: 'RNPM_PROCESS_QUARANTINED',
        status: 409,
      });
    }
    if (options.force) {
      group.quarantined = false;
      group.restartPolicy.markStable();
    }
    group.desiredRunning = true;
    if (options.instances != null) group.desiredInstances = Math.max(1, Number(options.instances) || 1);

    const created = [];
    for (let slot = 0; slot < group.desiredInstances; slot++) {
      const current = this.registry.active(group.name, slot);
      if (current && liveState(current.state.state)) continue;
      created.push(await this._spawn(group, slot, { activate: true }));
    }
    await Promise.all(created.map((record) => this._waitReady(record)));
    return this.registry.snapshotGroup(group.name);
  }

  async _spawn(group, slot, options = {}) {
    const definition = group.definition;
    const record = this.registry.createRecord(group.name, slot, {
      replacementFor: options.replacementFor || null,
    });
    record.suppressRestart = options.suppressRestart === true;
    record.waiters.set('ready', new Deferred());
    record.waiters.set('drained', new Deferred());
    record.waiters.set('shutdown', new Deferred());
    record.waiters.set('exit', new Deferred());
    record.state.transition('starting', { reason: options.replacementFor ? 'replacement' : 'start' });
    this.emit('process:starting', record);
    await this.journal?.append?.('process.starting', { id: record.id, name: record.name, slot, generation: record.generation });

    let handle;
    try {
      handle = await this.drivers[definition.mode].spawn(definition, {
        slot,
        generation: record.generation,
        replacementFor: options.replacementFor || null,
      });
    } catch (error) {
      record.error = serializeError(error, { stack: true });
      record.state.transition('boot_failed', { reason: 'spawn-error', details: record.error });
      record.waiters.get('ready').reject(error);
      throw error;
    }

    record.handle = handle;
    record.child = handle.child;
    record.worker = handle.worker;
    record.pid = handle.pid;
    record.startedAt = Date.now();
    record.lastHeartbeatAt = record.startedAt;
    this._attachHandle(record);
    this.logs?.attach?.(record);
    if (record.state.state === 'starting') record.state.transition('online', { reason: 'spawned' });
    this.emit('process:online', record);

    if (definition.readiness.mode === 'immediate') {
      queueMicrotask(() => this._markReady(record, { source: 'immediate' }));
    } else if (definition.readiness.mode === 'http' || definition.readiness.mode === 'tcp') {
      waitForProbe(definition.readiness)
        .then(() => this._markReady(record, { source: definition.readiness.mode }))
        .catch((error) => record.waiters.get('ready')?.reject(error));
    }

    if (options.activate === true && definition.readiness.mode === 'immediate') {
      this.registry.activate(record);
    }
    return record;
  }

  _attachHandle(record) {
    const handle = record.handle;
    handle.on('message', (message) => this._onRuntimeMessage(record, message));
    handle.on('listening', (address) => this.emit('process:listening', record, address));
    handle.on('error', (error) => {
      record.error = serializeError(error, { stack: true });
      this.emit('process:error', record, error);
    });
    handle.once('exit', (code, signal) => {
      this._onExit(record, code, signal).catch((error) => this.emit('error', error));
    });
  }

  _markReady(record, meta = {}) {
    if (!record || ['stopped', 'crashed', 'quarantined', 'boot_failed'].includes(record.state.state)) return;
    record.readyAt = record.readyAt || Date.now();
    if (record.state.state !== 'ready' && record.state.state !== 'running') {
      record.state.transition('ready', { reason: meta.source || 'runtime' });
      record.state.transition('running', { reason: 'accepted-traffic' });
    }
    record.waiters.get('ready')?.resolve(record);
    this.emit('process:ready', record, meta);
  }

  async _waitReady(record) {
    try {
      await withTimeout(record.waiters.get('ready').promise, record.definition.readiness.timeoutMs, {
        message: `Process '${record.name}' slot ${record.slot} did not become ready within ${record.definition.readiness.timeoutMs}ms.`,
        code: 'RNPM_READINESS_TIMEOUT',
      });
      if (!record.replacementFor && this.registry.active(record.name, record.slot)?.id !== record.id) this.registry.activate(record);
      return record;
    } catch (error) {
      record.error = serializeError(error, { stack: true });
      if (liveState(record.state.state)) record.state.transition('boot_failed', { reason: 'readiness-failed', details: record.error, force: true });
      record.expectedExit = true;
      await this._terminateRecord(record, { skipDrain: true }).catch(() => {});
      throw error;
    }
  }

  _onRuntimeMessage(record, message) {
    if (!isRuntimePacket(message)) return;
    record.hasRuntimeAgent = true;
    record.lastHeartbeatAt = Date.now();

    if (message.replyTo) {
      const pending = this.pendingRuntimeRequests.get(message.replyTo);
      if (pending) {
        this.pendingRuntimeRequests.delete(message.replyTo);
        clearTimeout(pending.timer);
        if (message.type === RUNTIME_MESSAGE.ERROR) {
          const error = new Error(message.payload?.error?.message || 'Runtime request failed.');
          error.code = message.payload?.error?.code || 'RNPM_RUNTIME_REQUEST_FAILED';
          pending.deferred.reject(error);
        } else pending.deferred.resolve(message.payload);
      }
    }

    switch (message.type) {
      case RUNTIME_MESSAGE.ONLINE:
        if (record.state.state === 'starting') record.state.transition('online', { reason: 'runtime-online' });
        this.emit('runtime:online', record, message.payload);
        break;
      case RUNTIME_MESSAGE.READY:
        this._markReady(record, { source: 'ipc', payload: message.payload });
        break;
      case RUNTIME_MESSAGE.HEARTBEAT:
        this.emit('runtime:heartbeat', record, message.payload);
        break;
      case RUNTIME_MESSAGE.HEALTH:
        record.lastHealth = message.payload;
        record.healthFailures = message.payload?.ok === false ? record.healthFailures + 1 : 0;
        this.emit('runtime:health', record, message.payload);
        break;
      case RUNTIME_MESSAGE.METRICS:
        record.metrics = message.payload;
        this.metrics?.update?.(record, message.payload);
        this.emit('runtime:metrics', record, message.payload);
        break;
      case RUNTIME_MESSAGE.DRAINING:
        if (record.state.can('draining')) record.state.transition('draining', { reason: 'runtime-draining' });
        break;
      case RUNTIME_MESSAGE.DRAINED:
        record.waiters.get('drained')?.resolve(message.payload);
        this.emit('runtime:drained', record, message.payload);
        break;
      case RUNTIME_MESSAGE.SHUTDOWN_COMPLETE:
        record.waiters.get('shutdown')?.resolve(message.payload);
        break;
      case RUNTIME_MESSAGE.ERROR:
        record.error = message.payload?.error || message.payload;
        this.emit('runtime:error', record, message.payload);
        break;
      case RUNTIME_MESSAGE.DIAGNOSTIC_READY:
        this.emit('runtime:diagnostic', record, message.payload);
        break;
      default:
        this.emit('runtime:message', record, message);
    }
  }

  async _requestRuntime(record, type, payload = {}, timeoutMs = 5_000) {
    if (!record.handle?.connected || !record.hasRuntimeAgent) {
      const error = new Error(`Process '${record.name}' slot ${record.slot} has no connected RawNode runtime agent.`);
      error.code = 'RNPM_RUNTIME_UNAVAILABLE';
      throw error;
    }
    const requestId = randomUUID();
    const deferred = new Deferred();
    const timer = setTimeout(() => {
      this.pendingRuntimeRequests.delete(requestId);
      const error = new Error(`Runtime request '${type}' timed out after ${timeoutMs}ms.`);
      error.code = 'RNPM_RUNTIME_TIMEOUT';
      deferred.reject(error);
    }, timeoutMs);
    timer.unref?.();
    this.pendingRuntimeRequests.set(requestId, { deferred, timer, record });
    try {
      await record.handle.send(runtimePacket(type, payload, { requestId }));
    } catch (error) {
      clearTimeout(timer);
      this.pendingRuntimeRequests.delete(requestId);
      deferred.reject(error);
    }
    return deferred.promise;
  }

  async _onExit(record, code, signal) {
    record.exitCode = code;
    record.signal = signal;
    record.stoppedAt = Date.now();
    if (!record.waiters.get('ready')?.settled) {
      const readinessError = new Error(`Process '${record.name}' slot ${record.slot} exited before readiness (code=${code}, signal=${signal || 'none'}).`);
      readinessError.code = 'RNPM_EXIT_BEFORE_READY';
      record.waiters.get('ready')?.reject(readinessError);
    }
    for (const [requestId, pending] of this.pendingRuntimeRequests) {
      if (pending.record.id !== record.id) continue;
      clearTimeout(pending.timer);
      pending.deferred.reject(new Error(`Process exited before runtime request completed.`));
      this.pendingRuntimeRequests.delete(requestId);
    }
    if (this.logs?.detach) await this.logs.detach(record).catch(() => {});
    this.metrics?.remove?.(record);
    this.registry.deactivate(record);

    const expected = record.expectedExit === true;
    if (expected) {
      if (record.state.state !== 'stopped') record.state.transition('stopped', { reason: 'expected-exit', force: true, details: { code, signal } });
      this.emit('process:stopped', record);
      record.waiters.get('exit')?.resolve({ code, signal });
      this.registry.prune(record.name);
      return;
    }

    record.state.transition('crashed', { reason: 'unexpected-exit', force: true, details: { code, signal } });
    this.emit('process:crashed', record);
    record.waiters.get('exit')?.resolve({ code, signal });
    this.registry.prune(record.name);
    const group = this.registry.get(record.name);
    if (!group || !group.desiredRunning || record.suppressRestart || record.slot >= group.desiredInstances) return;

    const decision = group.restartPolicy.decide({ expected, exitCode: code, signal, startedAt: record.startedAt });
    if (decision.quarantine) {
      group.quarantined = true;
      record.state.transition('quarantined', { reason: decision.reason, details: decision });
      this.emit('process:quarantined', record, decision);
      return;
    }
    if (!decision.restart) return;

    record.state.transition('backoff', { reason: decision.reason, details: decision });
    this.emit('process:backoff', record, decision);
    const timer = setTimeout(() => {
      this._lock(group.name).run(async () => {
        if (this.closed || !group.desiredRunning || group.quarantined || record.slot >= group.desiredInstances) return;
        const active = this.registry.active(group.name, record.slot);
        if (active && liveState(active.state.state)) return;
        const replacement = await this._spawn(group, record.slot, { activate: true });
        await this._waitReady(replacement);
      }).catch((error) => this.emit('error', error));
    }, decision.delayMs);
    timer.unref?.();
    record.timers.add(timer);
  }

  async stop(name, options = {}) {
    const group = this.registry.requireGroup(name);
    return this._lock(name).run(() => this._stopGroup(group, options));
  }

  async _stopGroup(group, options = {}) {
    group.desiredRunning = false;
    const records = this.registry.groupProcesses(group.name, { includeStopped: false });
    await mapLimit(records, options.concurrency || 8, (record) => this._drainAndStop(record, options));
    if (options.deleteDefinition) this.registry.deleteGroup(group.name);
    return options.deleteDefinition ? { name: group.name, deleted: true } : this.registry.snapshotGroup(group.name);
  }

  async _drainAndStop(record, options = {}) {
    if (!record || record.waiters.get('exit')?.settled || record.stoppedAt) return;
    record.expectedExit = true;
    const graceMs = Number(options.graceMs ?? record.definition.shutdown.graceMs);
    const killMs = Number(options.killMs ?? record.definition.shutdown.killMs);

    if (record.hasRuntimeAgent && !options.skipDrain) {
      if (record.state.can('draining')) record.state.transition('draining', { reason: options.reason || 'stop' });
      try {
        await record.handle.send(runtimePacket(SUPERVISOR_MESSAGE.DRAIN, { deadlineMs: graceMs }, { requestId: randomUUID() }));
        await withTimeout(record.waiters.get('drained').promise, graceMs, { message: `Drain timed out for ${record.name}:${record.slot}` });
      } catch (error) {
        this.emit('process:drain-timeout', record, error);
      }
      try {
        await record.handle.send(runtimePacket(SUPERVISOR_MESSAGE.SHUTDOWN, { deadlineMs: killMs }, { requestId: randomUUID() }));
      } catch {}
    } else if (!options.skipDrain && record.state.can('draining')) {
      record.state.transition('draining', { reason: 'signal-drain' });
    }

    if (record.state.can('stopping')) record.state.transition('stopping', { reason: options.reason || 'stop' });

    if (record.hasRuntimeAgent) {
      try {
        await withTimeout(record.waiters.get('exit').promise, killMs, { message: `Graceful shutdown timed out for ${record.name}:${record.slot}` });
        return;
      } catch {}
    }

    try { record.handle.kill(record.definition.shutdown.signal); } catch {}
    try {
      await withTimeout(record.waiters.get('exit').promise, killMs, { message: `${record.definition.shutdown.signal} timed out for ${record.name}:${record.slot}` });
      return;
    } catch {}

    try { record.handle.kill(record.definition.shutdown.killSignal); } catch {}
    await withTimeout(record.waiters.get('exit').promise, Math.max(1_000, killMs), {
      message: `Process ${record.name}:${record.slot} did not exit after ${record.definition.shutdown.killSignal}.`,
    }).catch(() => {});
  }

  async _terminateRecord(record, options = {}) {
    record.expectedExit = true;
    return this._drainAndStop(record, { ...options, reason: options.reason || 'terminate' });
  }

  async restart(name, options = {}) {
    const group = this.registry.requireGroup(name);
    return this._lock(name).run(async () => {
      await this._stopGroup(group, { ...options, deleteDefinition: false });
      group.desiredRunning = true;
      return this._startGroup(group, { ...options, force: options.force ?? true });
    });
  }

  async reload(name, options = {}) {
    const group = this.registry.requireGroup(name);
    if (group.definition.mode !== 'cluster') return this.restart(name, options);
    return this._lock(name).run(async () => {
      if (!group.desiredRunning) return this._startGroup(group, options);
      const oldRecords = this.registry.active(name);
      for (const oldRecord of oldRecords) {
        const replacement = await this._spawn(group, oldRecord.slot, {
          activate: false,
          replacementFor: oldRecord.id,
          suppressRestart: true,
        });
        try {
          await this._waitReady(replacement);
          this.registry.activate(replacement);
          replacement.suppressRestart = false;
          oldRecord.expectedExit = true;
          await this._drainAndStop(oldRecord, { ...options, reason: 'rolling-reload' });
          this.emit('process:reloaded', replacement, oldRecord);
        } catch (error) {
          replacement.expectedExit = true;
          await this._terminateRecord(replacement, { skipDrain: true, reason: 'failed-replacement' }).catch(() => {});
          this.emit('process:reload-failed', oldRecord, error);
          throw error;
        }
      }
      return this.registry.snapshotGroup(name);
    });
  }

  async scale(name, instances, options = {}) {
    const group = this.registry.requireGroup(name);
    const target = Math.max(1, Math.trunc(Number(instances) || 1));
    if (group.definition.mode !== 'cluster' && target !== 1) {
      throw new ProcessLifecycleError(`Only cluster mode can scale beyond one instance.`, { code: 'RNPM_SCALE_MODE_INVALID', status: 409 });
    }
    return this._lock(name).run(async () => {
      const previous = group.desiredInstances;
      group.desiredInstances = target;
      if (!group.desiredRunning) return this.registry.snapshotGroup(name);
      if (target > previous) {
        const records = [];
        for (let slot = previous; slot < target; slot++) records.push(await this._spawn(group, slot, { activate: true }));
        await Promise.all(records.map((record) => this._waitReady(record)));
      } else if (target < previous) {
        const records = this.registry.active(name).filter((record) => record.slot >= target).sort((a, b) => b.slot - a.slot);
        await mapLimit(records, options.concurrency || 4, (record) => {
          record.expectedExit = true;
          return this._drainAndStop(record, { ...options, reason: 'scale-down' });
        });
      }
      this.emit('process:scaled', group, previous, target);
      return this.registry.snapshotGroup(name);
    });
  }

  async diagnostics(name, options = {}) {
    const group = this.registry.requireGroup(name);
    const records = options.slot == null ? this.registry.active(name) : [this.registry.active(name, Number(options.slot))].filter(Boolean);
    const out = [];
    for (const record of records) {
      try {
        out.push({ id: record.id, pid: record.pid, result: await this._requestRuntime(record, SUPERVISOR_MESSAGE.DIAGNOSTICS, {}, options.timeoutMs || 10_000) });
      } catch (error) {
        out.push({ id: record.id, pid: record.pid, error: serializeError(error) });
      }
    }
    return { name: group.name, diagnostics: out };
  }

  async _healthTick() {
    if (this.closed) return;
    const now = Date.now();
    for (const group of this.registry.groups.values()) {
      if (!group.desiredRunning || group.quarantined) continue;
      for (const record of this.registry.active(group.name)) {
        if (!record || !['online', 'ready', 'running', 'unhealthy'].includes(record.state.state)) continue;
        const health = record.definition.health;
        if (!health.enabled || !record.hasRuntimeAgent) continue;

        if (now - record.lastHeartbeatAt > health.heartbeatTimeoutMs) {
          record.healthFailures++;
          record.lastHealth = { ok: false, error: 'heartbeat-timeout', lastHeartbeatAt: record.lastHeartbeatAt };
        } else if (!record.healthProbePending && (!record.lastHealthProbeAt || now - record.lastHealthProbeAt >= health.intervalMs)) {
          record.lastHealthProbeAt = now;
          record.healthProbePending = true;
          this._requestRuntime(record, SUPERVISOR_MESSAGE.HEALTH, {}, health.timeoutMs)
            .then((result) => {
              record.lastHealth = result;
              record.healthFailures = result?.ok === false ? record.healthFailures + 1 : 0;
            })
            .catch((error) => {
              record.lastHealth = { ok: false, error: error?.message || String(error) };
              record.healthFailures++;
            })
            .finally(() => { record.healthProbePending = false; });
        }

        if (record.healthFailures >= health.unhealthyThreshold && !record.healthRestarting) {
          record.healthRestarting = true;
          if (record.state.can('unhealthy')) record.state.transition('unhealthy', { reason: 'health-threshold', details: record.lastHealth });
          this.emit('process:unhealthy', record, record.lastHealth);
          if (health.restart) {
            this._lock(group.name).run(async () => {
              const active = this.registry.active(group.name, record.slot);
              if (!active || active.id !== record.id || !group.desiredRunning) return;
              if (group.definition.mode === 'cluster') {
                const replacement = await this._spawn(group, record.slot, { activate: false, replacementFor: record.id, suppressRestart: true });
                await this._waitReady(replacement);
                this.registry.activate(replacement);
                replacement.suppressRestart = false;
                record.expectedExit = true;
                await this._drainAndStop(record, { reason: 'unhealthy-replacement' });
              } else {
                record.expectedExit = true;
                await this._drainAndStop(record, { reason: 'unhealthy-restart' });
                const replacement = await this._spawn(group, record.slot, { activate: true });
                await this._waitReady(replacement);
              }
            }).catch((error) => this.emit('error', error)).finally(() => { record.healthRestarting = false; });
          } else record.healthRestarting = false;
        }
      }
    }
  }

  status(name = null) {
    return name ? this.registry.snapshotGroup(name) : this.registry.list();
  }

  async close(options = {}) {
    if (this.closed) return;
    this.closed = true;
    clearInterval(this._healthTimer);
    for (const record of this.registry.processes.values()) {
      for (const timer of record.timers || []) clearTimeout(timer);
      record.timers?.clear?.();
    }
    const groups = Array.from(this.registry.groups.values());
    await mapLimit(groups, options.concurrency || 4, async (group) => {
      group.desiredRunning = false;
      const records = this.registry.groupProcesses(group.name, { includeStopped: false });
      await mapLimit(records, 8, (record) => this._drainAndStop(record, { reason: 'supervisor-close' }));
    });
    for (const pending of this.pendingRuntimeRequests.values()) {
      clearTimeout(pending.timer);
      pending.deferred.reject(new Error('Supervisor closed.'));
    }
    this.pendingRuntimeRequests.clear();
  }
}

export default Supervisor;

  _wireEvents() {
    const publishRecord = (event) => (record, details = null) => {
      this.publish(event, { record: recordView(record), details });
    };
    for (const event of [
      'process:starting', 'process:online', 'process:ready', 'process:stopped',
      'process:crashed', 'process:backoff', 'process:quarantined', 'process:unhealthy',
      'process:reloaded', 'process:reload-failed', 'process:drain-timeout', 'process:error',

    this.supervisor.on('transition', (record, entry) => this.publish('process:transition', { record: recordView(record), entry }));
    this.supervisor.on('runtime:health', (record, health) => this.publish('runtime:health', { record: recordView(record), health }));
    this.supervisor.on('runtime:error', (record, error) => this.publish('runtime:error', { record: recordView(record), error }));
    this.supervisor.on('error', (error) => this.publish('daemon:error', { error: serializeError(error, { stack: true }) }));
    this.logs.on('line', (line) => this.publish('log:line', line));

    this.watcher.on('changes', (name, changes) => this.publish('watch:changes', { name, changes }));
    this.watcher.on('nativeError', (error, name, dir) => this.publish('watch:native-error', { name, dir, error: serializeError(error) }));
    this.watcher.on('error', (error, name) => this.publish('watch:error', { name, error: serializeError(error) }));
    this.control.on('error', (error) => this.publish('control:error', { error: serializeError(error, { stack: false }) }));
    this.control.on('serverError', (error) => this.publish('control:error', { error: serializeError(error, { stack: false }) }));
    this.realtime.on('event', (packet) => this.control.broadcast(packet.event, packet.payload));

  publish(event, payload = {}) {
    const packet = this.realtime.publish(event, payload);

  async start(options = {}) {
    if (this.startedAt) return this;
    if (this.closed) {
      const error = new Error('ProcessManager cannot be started after it has been closed.');
      error.code = 'RNPM_MANAGER_CLOSED';
      throw error;
    }

    try {
      await ensureDir(this.paths.root, 0o700);
      await ensureDir(this.paths.logs, 0o700);
      await ensureDir(this.paths.incidents, 0o700);
      await this.lock.acquire();
      await this.authenticator.initialize();
      await this.journal.open();
      await this.audit.open();
      await atomicWriteFile(this.paths.daemonPid, `${process.pid}\n`, { encoding: 'utf8', mode: 0o600 });
      await this.control.start();
      await this.management.start();
    } catch (error) {
      await this._rollbackFailedStart(error);
      throw error;
    }

    try {
      await this.journal.append('daemon.started', { pid: process.pid, socket: this.paths.socket });
      this.startedAt = Date.now();
      this.publish('daemon:started', { pid: process.pid, socket: this.paths.socket, stateDir: this.paths.root });
    } catch (error) {
      this.startedAt = null;
      await this._rollbackFailedStart(error);
      throw error;
    }

    if (options.resurrect === true) {
      try {
        await this.resurrect({ ignoreMissing: true });
      } catch (error) {
        const details = { error: serializeError(error, { stack: false }) };
        await this.journal.append('state.resurrect-failed', details).catch(() => {});

        if (options.failOnResurrectError === true) throw error;

  async handleCommand(command, params = {}, meta = {}) {

      case 'status': return this.supervisor.status(params.name || null);
      case 'show': return this.supervisor.status(params.name);
      case 'logs': return this.logs.query(params.name, { lines: params.lines, stream: params.stream });

  async startProcesses(params = {}) {
    let definitions;
    if (params.definition) {
      definitions = [ProcessDefinition.from(params.definition, { cwd: this.cwd })];
    } else {
      const loader = new ManifestLoader({ cwd: params.cwd || this.cwd });
      const manifest = await loader.load(params.manifest || null, { fresh: true });
      definitions = params.name ? manifest.apps.filter((app) => app.name === params.name) : manifest.apps.filter((app) => app.autostart || params.all === true);
      if (params.name && definitions.length === 0) {
        const error = new Error(`Process '${params.name}' was not found in ${manifest.file}.`);
        error.code = 'RNPM_PROCESS_NOT_IN_MANIFEST';
        throw error;
      }
    }

    const results = [];
    for (const definition of definitions) {
      const result = await this.supervisor.start(definition, { force: params.force === true, instances: params.instances });
      await this._syncWatch(definition.name);
      results.push(result);
    }
    if (this.autoSave) await this.saveState();
    return results.length === 1 ? results[0] : results;
  }

  async stopProcess(name, options = {}) {
    if (!name) throw new Error('stop requires process name.');
    await this.watcher.unwatch(name);

    if (this.autoSave) await this.saveState();
    return result;

    if (!name) throw new Error('restart requires process name.');

    if (this.autoSave) await this.saveState();
    return result;

    if (!name) throw new Error('reload requires process name.');
    const result = await this.supervisor.reload(name, options);
    if (this.autoSave) await this.saveState();
    return result;
  }

  async scaleProcess(name, instances, options = {}) {
    if (!name) throw new Error('scale requires process name.');
    const result = await this.supervisor.scale(name, instances, options);
    if (this.autoSave) await this.saveState();
    return result;
  }

  async deleteProcess(name, options = {}) {
    if (!name) throw new Error('delete requires process name.');
    await this.watcher.unwatch(name);
    const result = await this.supervisor.stop(name, { ...options, deleteDefinition: true });
    if (this.autoSave) await this.saveState();
    return result;

  async _syncWatch(name) {
    const group = this.registry.requireGroup(name);
    if (!group.desiredRunning || !group.definition.watch.enabled) {
      await this.watcher.unwatch(name);
      group.watchStatus = null;
      return;
    }
    await this.watcher.watch(group.definition, async (changes) => {
      this.publish('watch:action', { name, mode: group.definition.watch.mode, changes });
      if (!group.desiredRunning) return;
      if (group.definition.watch.mode === 'reload') await this.supervisor.reload(name, { reason: 'watch' });
      else if (group.definition.watch.mode === 'restart') await this.supervisor.restart(name, { reason: 'watch' });
      group.watchStatus = this.watcher.status(name);
      if (this.autoSave) await this.saveState();
    });
    group.watchStatus = this.watcher.status(name);
  }

  async saveState() {
    const state = {
      daemon: { cwd: this.cwd },
      desired: this.registry.desiredState(),
    };
    const saved = await this.stateStore.save(state);
    await this.journal.append('state.saved', { processCount: state.desired.processes.length });
    return { ok: true, file: this.paths.state, savedAt: saved.savedAt, processCount: state.desired.processes.length };
  }

  async resurrect(options = {}) {
    const state = await this.stateStore.load(null);
    if (!state) {
      if (options.ignoreMissing) return { ok: true, missing: true, processes: [] };
      const error = new Error('No saved RawNode PM state exists.');
      error.code = 'RNPM_STATE_MISSING';
      throw error;
    }
    const results = [];
    for (const item of state.desired?.processes || []) {
      const definition = ProcessDefinition.from(item.definition, { cwd: state.daemon?.cwd || this.cwd });
      const group = this.registry.define(definition);
      group.desiredInstances = Math.max(1, Number(item.desiredInstances) || definition.instances);
      group.quarantined = item.quarantined === true;
      if (item.desiredRunning !== false && !group.quarantined) {
        results.push(await this.supervisor.start(group.name, { force: options.force === true, instances: group.desiredInstances }));
        await this._syncWatch(group.name);
      }
    }
    await this.journal.append('state.resurrected', { processCount: results.length });
    return { ok: true, processes: results };
  }

  async collectDiagnostics(name, options = {}) {
    if (!name) throw new Error('diagnostics requires process name.');
    return this.diagnostics.collect(name, options);
  }

  uptimeMs() {
    return this.startedAt ? Date.now() - this.startedAt : 0;
  }

  health() {
    const groups = this.registry.list();
    const unhealthy = [];
    let activeProcesses = 0;
    for (const group of this.registry.groups.values()) {
      const active = this.registry.active(group.name);
      activeProcesses += active.length;
      if (group.quarantined) unhealthy.push({ name: group.name, state: 'quarantined' });
      for (const record of active) {
        if (['crashed', 'unhealthy', 'quarantined', 'boot_failed'].includes(record.state.state)) {
          unhealthy.push({ ...recordView(record) });
        }
      }
      if (group.desiredRunning && active.length < group.desiredInstances) {
        unhealthy.push({ name: group.name, state: 'capacity-mismatch', active: active.length, desired: group.desiredInstances });
      }
    }
    return {
      ok: !this.closed && unhealthy.length === 0,
      ready: !!this.startedAt && !this.closed,
      pid: process.pid,
      uptimeMs: this.uptimeMs(),
      socket: this.paths.socket,
      stateDir: this.paths.root,
      management: this.management.address(),
      groups: groups.length,
      activeProcesses,
      historyRecords: groups.reduce((sum, group) => sum + group.processes.length, 0),
      unhealthy,
    };
  }

  async doctor() {
    const issues = [];
    const nodeMajor = Number(process.versions.node.split('.')[0]);
    if (nodeMajor < 20) issues.push({ severity: 'error', code: 'NODE_VERSION', message: 'Node.js 20 or newer is required.' });
    for (const group of this.registry.list()) {
      if (group.quarantined) issues.push({ severity: 'error', code: 'PROCESS_QUARANTINED', process: group.name, message: 'Process is quarantined after repeated failures.' });
      if (group.desiredRunning && group.activeInstances < group.desiredInstances) issues.push({ severity: 'warning', code: 'CAPACITY_MISMATCH', process: group.name, message: `${group.activeInstances}/${group.desiredInstances} active instances.` });
      if (group.definition?.mode === 'fork' && group.desiredInstances > 1) issues.push({ severity: 'error', code: 'FORK_SCALE_INVALID', process: group.name, message: 'Fork mode cannot own multiple instances.' });
    }
    return {
      ok: !issues.some((issue) => issue.severity === 'error'),
      node: process.version,
      platform: process.platform,
      pid: process.pid,
      paths: { ...this.paths },
      health: this.health(),
      issues,
    };
  }

  async close() {
    if (this.closed) return;
    this.closed = true;

    const errors = [];
    try { this.publish('daemon:stopping', { pid: process.pid }); }
    catch (error) { errors.push({ name: 'daemon.stopping.publish', error }); }

    const attempt = async (name, fn) => {
      try { await fn(); }
      catch (error) { errors.push({ name, error }); }
    };

    await attempt('watcher.close', () => this.watcher.close());
    await attempt('supervisor.close', () => this.supervisor.close());
    await attempt('logs.close', () => this.logs.close());
    await attempt('journal.stopped', () => this.journal.append('daemon.stopped', { pid: process.pid }));
    await attempt('audit.stopped', () => this.audit.record('daemon.stopped', { pid: process.pid }));
    await attempt('management.close', () => this.management.close());
    await attempt('control.close', () => this.control.close());
    await attempt('journal.close', () => this.journal.close());
    await attempt('audit.close', () => this.audit.close());
    await attempt('daemon.pid.remove', () => removeIfExists(this.paths.daemonPid));
    await attempt('daemon.lock.release', () => this.lock.release());

    this.emit('closed', errors.map(({ name, error }) => ({ name, error: serializeError(error, { stack: false }) })));

    if (errors.length) {
      throw new AggregateError(
        errors.map(({ error }) => error),
        `RawNode PM daemon shutdown completed with ${errors.length} cleanup error(s).`,
      );
    }
  }
}

export default ProcessManager;

'use strict';

function processSnapshot(record) {
  const now = Date.now();
  return {
    id: record.id,
    name: record.name,
    slot: record.slot,
    generation: record.generation,
    pid: record.pid || null,
    state: record.state.state,
    stateChangedAt: record.state.changedAt,
    startedAt: record.startedAt || null,
    readyAt: record.readyAt || null,
    stoppedAt: record.stoppedAt || null,
    uptimeMs: record.startedAt && !record.stoppedAt ? now - record.startedAt : 0,
    expectedExit: record.expectedExit === true,
    exitCode: record.exitCode ?? null,
    signal: record.signal ?? null,
    lastHeartbeatAt: record.lastHeartbeatAt || null,
    lastHealth: record.lastHealth || null,
    metrics: record.metrics || null,
    error: record.error || null,
    replacementFor: record.replacementFor || null,
  };
}

export class ProcessRegistry extends EventEmitter {
  constructor(options = {}) {
    super();
    this.historyPerGroup = Math.max(10, Number(options.historyPerGroup) || 100);
    this.groups = new Map();
    this.processes = new Map();
  }

  define(input) {
    const definition = ProcessDefinition.from(input);
    let group = this.groups.get(definition.name);
    if (!group) {
      group = {
        name: definition.name,
        definition,
        desiredInstances: definition.instances,
        desiredRunning: false,
        generation: 0,
        activeBySlot: new Map(),
        processIds: new Set(),
        restartPolicy: new RestartPolicy(definition.restart),
        quarantined: false,
        watchStatus: null,
        createdAt: Date.now(),
        updatedAt: Date.now(),
      };
      this.groups.set(definition.name, group);
      this.emit('defined', group);
    } else {
      group.definition = definition;
      group.desiredInstances = definition.instances;
      group.updatedAt = Date.now();
      group.restartPolicy = new RestartPolicy(definition.restart);
      this.emit('updated', group);
    }
    return group;
  }

  createRecord(name, slot, options = {}) {
    const group = this.requireGroup(name);
    const generation = options.generation ?? ++group.generation;
    const id = options.id || `${name}:${slot}:${generation}:${randomUUID().slice(0, 8)}`;
    const record = {
      id,
      name,
      slot,
      generation,
      definition: group.definition,
      state: new ProcessStateMachine('defined'),
      handle: null,
      child: null,
      worker: null,
      pid: null,
      startedAt: null,
      readyAt: null,
      stoppedAt: null,
      expectedExit: false,
      exitCode: null,
      signal: null,
      lastHeartbeatAt: null,
      lastHealth: null,
      healthFailures: 0,
      metrics: null,
      error: null,
      replacementFor: options.replacementFor || null,
      waiters: new Map(),
      timers: new Set(),
    };
    record.state.on('transition', (entry) => this.emit('transition', record, entry));
    this.processes.set(id, record);
    group.processIds.add(id);
    this.emit('processCreated', record);
    return record;
  }

  activate(record) {
    const group = this.requireGroup(record.name);
    const previousId = group.activeBySlot.get(record.slot) || null;
    group.activeBySlot.set(record.slot, record.id);
    group.updatedAt = Date.now();
    this.emit('activated', record, previousId ? this.processes.get(previousId) : null);
    return previousId ? this.processes.get(previousId) : null;
  }

  active(name, slot = null) {
    const group = this.groups.get(name);
    if (!group) return slot == null ? [] : null;
    if (slot != null) {
      const id = group.activeBySlot.get(slot);
      return id ? this.processes.get(id) || null : null;
    }
    return Array.from(group.activeBySlot.entries())
      .sort((a, b) => a[0] - b[0])
      .map(([, id]) => this.processes.get(id))
      .filter(Boolean);
  }

  groupProcesses(name, { includeStopped = true } = {}) {
    const group = this.groups.get(name);
    if (!group) return [];
    const records = Array.from(group.processIds, (id) => this.processes.get(id)).filter(Boolean);
    return includeStopped ? records : records.filter((record) => !['stopped', 'crashed', 'quarantined', 'boot_failed'].includes(record.state.state));
  }

  deactivate(record) {
    const group = this.groups.get(record.name);
    if (!group) return;
    if (group.activeBySlot.get(record.slot) === record.id) group.activeBySlot.delete(record.slot);
    group.updatedAt = Date.now();
  }

  prune(name, limit = this.historyPerGroup) {
    const group = this.groups.get(name);
    if (!group) return 0;
    const activeIds = new Set(group.activeBySlot.values());
    const records = Array.from(group.processIds, (id) => this.processes.get(id)).filter(Boolean);
    const terminal = records
      .filter((record) => !activeIds.has(record.id) && ['stopped', 'crashed', 'quarantined', 'boot_failed'].includes(record.state.state))
      .sort((a, b) => b.generation - a.generation);
    let removed = 0;
    for (const record of terminal.slice(Math.max(0, Number(limit) || this.historyPerGroup))) {
      group.processIds.delete(record.id);
      this.processes.delete(record.id);
      removed++;
    }
    return removed;
  }

  deleteGroup(name) {
    const group = this.groups.get(name);
    if (!group) return false;
    for (const id of group.processIds) this.processes.delete(id);
    this.groups.delete(name);
    this.emit('deleted', name);
    return true;
  }

  get(name) { return this.groups.get(name) || null; }
  requireGroup(name) {
    const group = this.get(name);
    if (!group) {
      const error = new Error(`Unknown process '${name}'.`);
      error.code = 'RNPM_PROCESS_NOT_FOUND';
      throw error;
    }
    return group;
  }

  list() {
    return Array.from(this.groups.values())
      .sort((a, b) => a.name.localeCompare(b.name))
      .map((group) => this.snapshotGroup(group.name));
  }

  snapshotGroup(name) {
    const group = this.requireGroup(name);
    const processes = this.groupProcesses(name).sort((a, b) => a.slot - b.slot || b.generation - a.generation).map(processSnapshot);
    return {
      name: group.name,
      namespace: group.definition.namespace,
      mode: group.definition.mode,
      desiredInstances: group.desiredInstances,
      desiredRunning: group.desiredRunning === true,
      activeInstances: group.activeBySlot.size,
      quarantined: group.quarantined,
      createdAt: group.createdAt,
      updatedAt: group.updatedAt,
      watch: group.watchStatus,
      restart: group.restartPolicy.snapshot(),
      definition: group.definition.toJSON({ includeEnv: false }),
      processes,
    };
  }

  desiredState() {
    return {
      processes: Array.from(this.groups.values()).map((group) => ({
        definition: group.definition.toJSON({ includeEnv: true }),
        desiredInstances: group.desiredInstances,
        desiredRunning: group.desiredRunning === true,
        quarantined: group.quarantined,
      })),
    };
  }
}

export { processSnapshot };
export default ProcessRegistry;

'use strict';

import { EventEmitter } from 'node:events';

export class RealtimeGateway extends EventEmitter {
  constructor(options = {}) {
    super();
    this.historyLimit = Math.max(10, Number(options.historyLimit) || 1_000);
    this.history = [];
    this.adapters = new Set();
  }

  publish(event, payload = {}) {
    const packet = { event: String(event), payload, at: Date.now() };
    this.history.push(packet);
    if (this.history.length > this.historyLimit) this.history.splice(0, this.history.length - this.historyLimit);
    this.emit('event', packet);
    this.emit(packet.event, packet.payload);
    for (const adapter of this.adapters) {
      try {
        Promise.resolve(adapter.publish(packet)).catch((error) => this.emit('adapterError', error, adapter));
      } catch (error) {
        this.emit('adapterError', error, adapter);
      }
    }
    return packet;
  }

  attach(adapter) {
    if (!adapter || typeof adapter.publish !== 'function') throw new Error('Realtime adapter requires publish(packet).');
    this.adapters.add(adapter);
    return () => this.adapters.delete(adapter);
  }

  attachRawNodeWebSocket(io, options = {}) {
    const namespaceName = options.namespace || '/__rawnode/pm';
    const namespace = typeof io?.of === 'function' ? io.of(namespaceName) : io;
    return this.attach({
      publish(packet) {
        if (typeof namespace?.emit === 'function') namespace.emit('pm:event', packet);
        else if (typeof namespace?.broadcast === 'function') namespace.broadcast('pm:event', packet);
      },
    });
  }

  recent(options = {}) {
    const event = options.event || null;
    const limit = Math.max(1, Number(options.limit) || 100);
    const values = event ? this.history.filter((entry) => entry.event === event) : this.history;
    return values.slice(-limit);
  }
}

export default RealtimeGateway;

        await atomicWriteFile(edit.file, edit.source, { encoding: 'utf8', mode: 0o644 });
        written.push(edit.file);
      }
      for (const entry of plan.creates) {

          const backup = `${entry.file}.rawnode-pm.${stamp}.bak`;
          await atomicWriteFile(backup, original, { encoding: 'utf8', mode: 0o600 });
          backups.push({ file: entry.file, backup, original });

        await atomicWriteFile(entry.file, content, { encoding: 'utf8', mode: 0o644 });
        written.push(entry.file);
      }
      return {
        ok: true,
        dryRun: false,
        changed: plan.edits.filter((entry) => entry.changed).map((entry) => entry.file),
        created: plan.creates.map((entry) => entry.file),
        backups: backups.map((entry) => entry.backup),
      };
    } catch (error) {
      for (const backup of backups.reverse()) {
        await atomicWriteFile(backup.file, backup.original, { encoding: 'utf8', mode: 0o644 }).catch(() => {});
      }
      for (const file of created) await rm(file, { force: true }).catch(() => {});
      error.details = { ...(error.details || {}), rolledBack: written };
      throw error;
    }
  }

      serve = { source: result.source, changed: serve.changed || result.changed };

    if (options.dryRun === true) {
      return {
        ok: true,
        dryRun: true,
        changed: plan.edits.filter((entry) => entry.changed).map((entry) => entry.file),
        created: plan.creates.map((entry) => entry.file),
      };
    }

    const stamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backups = [];
    const created = [];
    const written = [];
    try {
      for (const edit of plan.edits) {
        if (!edit.changed) continue;
        const backup = `${edit.file}.rawnode-pm.${stamp}.bak`;
        await atomicWriteFile(backup, edit.original, { encoding: 'utf8', mode: 0o600 });
        backups.push({ file: edit.file, backup, original: edit.original });
        await atomicWriteFile(edit.file, edit.source, { encoding: 'utf8', mode: 0o644 });
        written.push(edit.file);
      }
      for (const entry of plan.creates) {
        const content = await readFile(entry.sourceFile, 'utf8');
        if (entry.existed) {
          const original = await readFile(entry.file, 'utf8');
          const backup = `${entry.file}.rawnode-pm.${stamp}.bak`;
          await atomicWriteFile(backup, original, { encoding: 'utf8', mode: 0o600 });
          backups.push({ file: entry.file, backup, original });
        } else created.push(entry.file);
        await atomicWriteFile(entry.file, content, { encoding: 'utf8', mode: 0o644 });
        written.push(entry.file);
      }
      return {
        ok: true,
        dryRun: false,
        changed: plan.edits.filter((entry) => entry.changed).map((entry) => entry.file),
        created: plan.creates.map((entry) => entry.file),
        backups: backups.map((entry) => entry.backup),
      };
    } catch (error) {
      for (const backup of backups.reverse()) {
        await atomicWriteFile(backup.file, backup.original, { encoding: 'utf8', mode: 0o644 }).catch(() => {});
      }
      for (const file of created) await rm(file, { force: true }).catch(() => {});
      error.details = { ...(error.details || {}), rolledBack: written };
      throw error;
    }
  }
}

class BasePmCommand {
  constructor(app) {
    this.app = app;
    this.signature = '';
    this.description = '';
    this.args = Object.create(null);
    this.flags = Object.create(null);
  }

  parse(argv = []) {
    const parts = this.signature.trim().split(/\s+/);
    if (parts[0] !== argv[0]) return false;
    const positionals = [];
    for (let index = 1; index < argv.length; index++) {
      const value = argv[index];
      if (value.startsWith('--')) {
        const [key, item] = value.slice(2).split('=');
        this.flags[key] = item ?? true;
      } else positionals.push(value);
    }
    let cursor = 0;
    for (const part of parts.slice(1)) if (part.startsWith('{') && part.endsWith('}')) this.args[part.slice(1, -1)] = positionals[cursor++];
    return true;
  }

  argument(name) { return this.args[name]; }
  option(name, fallback = undefined) { return this.flags[name] ?? fallback; }

    const paths = resolveStatePaths(configGet(this.app, 'pm.stateDir', process.env.RAWNODE_PM_HOME));
    return new ControlClient({ socketPath: paths.socket, tokenPath: paths.token });
  }

  output(value) {
    if (this.option('json') === true) console.log(JSON.stringify(value, null, 2));

  }

function systemdQuote(value) {
  return `"${String(value).replace(/\\/g, '\\\\').replace(/"/g, '\\"')}"`;
}

function xml(value) {
  return String(value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&apos;');
}

function plistString(value, indent = '    ') {
  return `${indent}<string>${xml(value)}</string>`;
}

function normalizePlatform(value) {
  const platform = String(value || '').trim().toLowerCase();
  if (platform === 'linux' || platform === 'systemd') return 'systemd';
  if (platform === 'darwin' || platform === 'macos' || platform === 'launchd') return 'launchd';
  return platform;
}

export class StartupGenerator {
  constructor(options = {}) {
    this.nodePath = path.resolve(options.nodePath || process.execPath);
    this.binPath = path.resolve(options.binPath);
    this.cwd = path.resolve(options.cwd || process.cwd());
    this.paths = resolveStatePaths(options.stateDir);
    this.user = options.user || null;
    this.group = options.group || null;
    this.userService = options.userService === true;
    this.label = String(options.label || 'org.rawnode.pm');
    this.description = String(options.description || 'RawNode PM Runtime Control Plane');
    this.resurrect = options.resurrect !== false;
    this.environment = {
      NODE_ENV: options.nodeEnv || process.env.NODE_ENV || 'production',
      RAWNODE_PM_HOME: this.paths.root,
      ...(options.environment || {}),
    };
    if (!options.binPath) throw new TypeError('StartupGenerator requires binPath.');
  }

  arguments() {
    const args = [this.nodePath, this.binPath, 'daemon', '--state-dir', this.paths.root, '--cwd', this.cwd];
    if (this.resurrect) args.push('--resurrect');
    return args;
  }

  systemd() {
    const serviceTarget = this.userService ? 'default.target' : 'multi-user.target';
    const lines = [
      '[Unit]',
      `Description=${this.description}`,
      'Wants=network-online.target',
      'After=network-online.target',
      '',
      '[Service]',
      'Type=simple',
      `WorkingDirectory=${systemdQuote(this.cwd)}`,
      `ExecStart=${this.arguments().map(systemdQuote).join(' ')}`,
      'ExecReload=/bin/kill -HUP $MAINPID',
      'Restart=on-failure',
      'RestartSec=2s',
      'TimeoutStartSec=30s',
      'TimeoutStopSec=45s',
      'KillMode=mixed',
      'UMask=0077',
      'LimitNOFILE=65535',
      'NoNewPrivileges=true',
    ];

    if (!this.userService && this.group) lines.splice(this.user ? 9 : 8, 0, `Group=${this.group}`);
    for (const [key, value] of Object.entries(this.environment)) {
      if (value == null) continue;
      lines.push(`Environment=${systemdQuote(`${key}=${value}`)}`);
    }
    lines.push('', '[Install]', `WantedBy=${serviceTarget}`, '');
    return lines.join('\n');
  }

  launchd() {
    const args = this.arguments().map((value) => plistString(value, '      ')).join('\n');
    const env = Object.entries(this.environment)
      .filter(([, value]) => value != null)
      .map(([key, value]) => `      <key>${xml(key)}</key>\n${plistString(value, '      ')}`)
      .join('\n');
    const outLog = path.join(this.paths.root, 'daemon-output.log');
    const errorLog = path.join(this.paths.root, 'daemon-error.log');
    return `<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>

  <array>
${args}
  </array>
  <key>WorkingDirectory</key>
  <string>${xml(this.cwd)}</string>
  <key>EnvironmentVariables</key>
  <dict>
${env}
  </dict>
  <key>RunAtLoad</key>
  <true/>
  <key>KeepAlive</key>
  <dict>
    <key>SuccessfulExit</key>
    <false/>
  </dict>
  <key>ProcessType</key>
  <string>Background</string>
  <key>ThrottleInterval</key>
  <integer>2</integer>
  <key>StandardOutPath</key>
  <string>${xml(outLog)}</string>
  <key>StandardErrorPath</key>
  <string>${xml(errorLog)}</string>
</dict>
</plist>
`;
  }

  generate(platform = process.platform) {
    const normalized = normalizePlatform(platform);
    if (normalized === 'systemd') {
      return {
        platform: normalized,
        filename: 'rawnode-pm.service',
        content: this.systemd(),

          ? path.join(process.env.HOME || '~', '.config/systemd/user/rawnode-pm.service')

    }
    if (normalized === 'launchd') {
      return {
        platform: normalized,
        filename: `${this.label}.plist`,
        content: this.launchd(),
        installTarget: path.join(process.env.HOME || '~', 'Library/LaunchAgents', `${this.label}.plist`),
      };
    }
    const error = new Error(`Unsupported startup platform '${platform}'. Use systemd or launchd.`);
    error.code = 'RNPM_STARTUP_PLATFORM_UNSUPPORTED';
    throw error;
  }

'use strict';

export class LineRingBuffer {
  constructor(options = {}) {
    this.maxBytes = Math.max(64 * 1024, Number(options.maxBytes) || 2 * 1024 * 1024);
    this.maxEntries = Math.max(100, Number(options.maxEntries) || 20_000);
    this.entries = [];
    this.bytes = 0;
  }

  push(entry) {
    const value = { ...entry, line: String(entry.line ?? '') };
    const size = Buffer.byteLength(value.line) + 128;
    value._bytes = size;
    this.entries.push(value);
    this.bytes += size;
    while (this.entries.length > this.maxEntries || this.bytes > this.maxBytes) {
      const removed = this.entries.shift();
      this.bytes -= removed?._bytes || 0;
    }
  }

  query(options = {}) {
    const lines = Math.max(1, Number(options.lines) || 100);
    const stream = options.stream || null;
    let values = stream ? this.entries.filter((entry) => entry.stream === stream) : this.entries;
    values = values.slice(-lines);
    return values.map(({ _bytes, ...entry }) => entry);
  }
}

export default LineRingBuffer;

'use strict';

import { createWriteStream } from 'node:fs';
import { stat, rename, rm } from 'node:fs/promises';
import path from 'node:path';
import { ensureDir } from '../support/fs.js';
import { AsyncLock } from '../support/async.js';

export class LogWriter {
  constructor(options = {}) {
    this.file = options.file;
    this.maxBytes = Math.max(64 * 1024, Number(options.maxBytes) || 50 * 1024 * 1024);
    this.maxFiles = Math.max(1, Number(options.maxFiles) || 5);
    this.mode = options.mode || 0o600;
    this.bytes = 0;
    this.stream = null;
    this.lock = new AsyncLock();
    this.closed = false;
  }

  async open() {
    if (this.stream) return this.stream;
    if (this.closed) throw new Error(`LogWriter for '${this.file}' is closed.`);
    await ensureDir(path.dirname(this.file), 0o700);
    try { this.bytes = (await stat(this.file)).size; } catch { this.bytes = 0; }
    this.stream = createWriteStream(this.file, { flags: 'a', mode: this.mode });
    return this.stream;
  }

  async write(chunk) {
    const buffer = Buffer.isBuffer(chunk) ? chunk : Buffer.from(String(chunk));
    return this.lock.run(async () => {
      if (this.closed) return false;
      await this.open();
      if (this.bytes > 0 && this.bytes + buffer.length > this.maxBytes) await this.rotate();
      await new Promise((resolve, reject) => {
        const stream = this.stream;
        const onError = (error) => { cleanup(); reject(error); };
        const onDrain = () => { cleanup(); resolve(); };
        const cleanup = () => {
          stream.off('error', onError);
          stream.off('drain', onDrain);
        };
        stream.once('error', onError);
        if (stream.write(buffer)) { cleanup(); resolve(); }
        else stream.once('drain', onDrain);
      });
      this.bytes += buffer.length;
      return true;
    });
  }

  async rotate() {
    const stream = this.stream;
    this.stream = null;
    if (stream) await new Promise((resolve) => stream.end(resolve));
    await rm(`${this.file}.${this.maxFiles}`, { force: true }).catch(() => {});
    for (let index = this.maxFiles - 1; index >= 1; index--) {
      await rename(`${this.file}.${index}`, `${this.file}.${index + 1}`).catch((error) => {
        if (error?.code !== 'ENOENT') throw error;
      });
    }
    await rename(this.file, `${this.file}.1`).catch((error) => {
      if (error?.code !== 'ENOENT') throw error;
    });
    this.bytes = 0;
    await this.open();
  }

  async close() {
    return this.lock.run(async () => {
      this.closed = true;
      const stream = this.stream;
      this.stream = null;
      if (stream) await new Promise((resolve) => stream.end(resolve));
    });
  }
}

export default LogWriter;

'use strict';

import { EventEmitter } from 'node:events';
import path from 'node:path';
import { LogWriter } from './LogWriter.js';
import { LineRingBuffer } from './LineRingBuffer.js';
import { safeName } from '../support/paths.js';

function timestampLine(prefix, chunk, enabled) {
  if (!enabled) return Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk);
  const text = Buffer.isBuffer(chunk) ? chunk.toString('utf8') : String(chunk);
  const stamp = new Date().toISOString();
  return Buffer.from(text.split(/(?<=\n)/).map((part) => part ? `${stamp} ${prefix}${part}` : '').join(''));
}

export class LogMultiplexer extends EventEmitter {
  constructor(options = {}) {
    super();
    this.root = path.resolve(options.root || 'storage/logs/pm');
    this.buffers = new Map();
    this.attachments = new Map();
  }

  _buffer(name, maxBytes) {
    let buffer = this.buffers.get(name);
    if (!buffer) {
      buffer = new LineRingBuffer({ maxBytes });
      this.buffers.set(name, buffer);
    }
    return buffer;
  }

  attach(record) {
    const handle = record.handle;
    if (!handle || record.definition.logs.enabled === false) return null;
    const name = safeName(record.name);
    const slot = Number(record.slot) || 0;
    const dir = path.join(this.root, name);
    const config = record.definition.logs;
    const writers = {
      stdout: new LogWriter({ file: path.join(dir, `${slot}-out.log`), maxBytes: config.maxBytes, maxFiles: config.maxFiles }),
      stderr: new LogWriter({ file: path.join(dir, `${slot}-error.log`), maxBytes: config.maxBytes, maxFiles: config.maxFiles }),
    };
    const partial = { stdout: '', stderr: '' };
    const listeners = [];
    const buffer = this._buffer(record.name, config.bufferBytes);

    const bind = (source, streamName) => {
      if (!source?.on) return;
      const onData = async (chunk) => {
        source.pause?.();
        try {
          const prefixed = timestampLine(`[${record.name}:${slot}:${record.pid || '?'}] `, chunk, config.timestamp);
          await writers[streamName].write(prefixed);
          this._captureLines(record, streamName, chunk, partial, buffer);
        } catch (error) {
          this.emit('error', error, record);
        } finally {
          source.resume?.();
        }
      };
      source.on('data', onData);
      listeners.push(() => source.off('data', onData));
    };

    bind(handle.stdout, 'stdout');
    bind(handle.stderr, 'stderr');
    const attachment = { record, writers, listeners, partial };
    this.attachments.set(record.id, attachment);
    return attachment;
  }

  _captureLines(record, stream, chunk, partial, buffer) {
    const text = partial[stream] + (Buffer.isBuffer(chunk) ? chunk.toString('utf8') : String(chunk));
    const values = text.split(/\r?\n/);
    partial[stream] = values.pop() || '';
    for (const line of values) {
      const entry = { at: Date.now(), name: record.name, slot: record.slot, pid: record.pid, stream, line };
      buffer.push(entry);
      this.emit('line', entry);
    }
  }

  query(name, options = {}) {
    return this.buffers.get(name)?.query(options) || [];
  }

  async detach(record) {
    const attachment = this.attachments.get(record.id);
    if (!attachment) return;
    this.attachments.delete(record.id);
    for (const off of attachment.listeners) off();
    for (const [stream, value] of Object.entries(attachment.partial)) {
      if (value) {
        const entry = { at: Date.now(), name: record.name, slot: record.slot, pid: record.pid, stream, line: value };
        this._buffer(record.name, record.definition.logs.bufferBytes).push(entry);
        this.emit('line', entry);
      }
    }
    await Promise.all(Object.values(attachment.writers).map((writer) => writer.close()));
  }

  async close() {
    await Promise.all(Array.from(this.attachments.values(), async (attachment) => {
      for (const off of attachment.listeners) off();
      await Promise.all(Object.values(attachment.writers).map((writer) => writer.close()));
    }));
    this.attachments.clear();
  }
}

const UNITS = Object.freeze({ ms: 1, s: 1_000, m: 60_000, h: 3_600_000, d: 86_400_000 });

export function durationMs(value, fallback = 0) {
  if (value == null || value === '') return fallback;
  if (typeof value === 'number') return Number.isFinite(value) ? Math.max(0, value) : fallback;
  const match = /^\s*(\d+(?:\.\d+)?)\s*(ms|s|m|h|d)?\s*$/i.exec(String(value));
  if (!match) return fallback;
  return Math.max(0, Number(match[1]) * (UNITS[match[2]?.toLowerCase() || 'ms'] || 1));
}

export function formatDuration(ms) {
  const n = Math.max(0, Number(ms) || 0);
  if (n < 1_000) return `${Math.round(n)}ms`;
  if (n < 60_000) return `${(n / 1_000).toFixed(n < 10_000 ? 1 : 0)}s`;
  if (n < 3_600_000) return `${Math.floor(n / 60_000)}m ${Math.floor((n % 60_000) / 1_000)}s`;
  return `${Math.floor(n / 3_600_000)}h ${Math.floor((n % 3_600_000) / 60_000)}m`;
}

'use strict';

export function sleep(ms, signal = null) {
  const duration = Math.max(0, Number(ms) || 0);
  if (signal?.aborted) return Promise.reject(signal.reason || new Error('Aborted'));
  return new Promise((resolve, reject) => {
    let settled = false;
    const cleanup = () => {
      if (signal) signal.removeEventListener('abort', onAbort);
    };
    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      cleanup();
      resolve();
    }, duration);
    const onAbort = () => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      cleanup();
      reject(signal.reason || new Error('Aborted'));
    };
    if (signal) signal.addEventListener('abort', onAbort, { once: true });
  });
}

export async function withTimeout(promiseOrFactory, timeoutMs, options = {}) {
  const ms = Number(timeoutMs) || 0;
  if (ms <= 0) return typeof promiseOrFactory === 'function' ? promiseOrFactory() : promiseOrFactory;
  let timer;
  const timeout = new Promise((_, reject) => {
    timer = setTimeout(() => {
      const error = new Error(options.message || `Operation timed out after ${ms}ms`);
      error.code = options.code || 'RNPM_TIMEOUT';
      reject(error);
    }, ms);
  });
  try {
    const task = typeof promiseOrFactory === 'function' ? Promise.resolve().then(promiseOrFactory) : promiseOrFactory;
    return await Promise.race([task, timeout]);
  } finally {
    clearTimeout(timer);
  }
}

export class Deferred {
  constructor() {
    this.settled = false;
    this.promise = new Promise((resolve, reject) => {
      this._resolve = resolve;
      this._reject = reject;
    });
    this.promise.catch(() => {});
  }
  resolve(value) { if (!this.settled) { this.settled = true; this._resolve(value); } }
  reject(error) { if (!this.settled) { this.settled = true; this._reject(error); } }
}

export class AsyncLock {
  constructor() { this._tail = Promise.resolve(); }
  run(fn) {
    const next = this._tail.then(fn, fn);
    this._tail = next.catch(() => {});
    return next;
  }
}

export async function mapLimit(values, limit, worker) {
  const list = Array.from(values || []);
  const out = new Array(list.length);
  let cursor = 0;
  const concurrency = Math.max(1, Math.min(list.length || 1, Number(limit) || 1));
  await Promise.all(Array.from({ length: concurrency }, async () => {
    while (cursor < list.length) {
      const index = cursor++;
      out[index] = await worker(list[index], index);
    }
  }));
  return out;
}

'use strict';

import path from 'node:path';

function escapeRegex(char) {
  return /[\\^$.*+?()[\]{}|]/.test(char) ? `\\${char}` : char;
}

export function globToRegExp(glob) {
  const input = String(glob || '').replaceAll('\\', '/');
  let out = '^';
  for (let i = 0; i < input.length; i++) {
    const char = input[i];
    if (char === '*') {
      if (input[i + 1] === '*') {
        i++;
        if (input[i + 1] === '/') { i++; out += '(?:.*/)?'; }
        else out += '.*';
      } else out += '[^/]*';
      continue;
    }
    if (char === '?') { out += '[^/]'; continue; }
    out += escapeRegex(char);
  }
  return new RegExp(`${out}$`);
}

export class IgnoreMatcher {
  constructor(patterns = []) {
    this.patterns = (Array.isArray(patterns) ? patterns : [patterns])
      .filter(Boolean)
      .map((pattern) => ({ pattern: String(pattern), re: globToRegExp(String(pattern)) }));
  }

  matches(filePath, root = process.cwd()) {
    const absolute = path.resolve(filePath);
    const relative = path.relative(root, absolute).replaceAll(path.sep, '/');
    const base = path.basename(absolute);
    return this.patterns.some(({ re }) => re.test(relative) || re.test(base) || re.test(absolute.replaceAll(path.sep, '/')));
  }
}

async function importAesKey(keyBytes) {
  return subtle.importKey('raw', keyBytes, { name: 'AES-GCM' }, false, ['encrypt', 'decrypt']);

export class JsonStateStore {
  constructor(options = {}) {
    this.file = options.file;
    this.keyProvider = options.keyProvider || null;
    this.encrypted = options.encrypted !== false;
  }

  async save(state) {

      schemaVersion: STATE_SCHEMA_VERSION,
      savedAt: new Date().toISOString(),
      ...state,
    };

    let envelope;
    if (this.encrypted && this.keyProvider) {
      const keyBytes = await this.keyProvider();
      const key = await importAesKey(keyBytes);
      const iv = randomBytes(12);
      const plaintext = encoder.encode(JSON.stringify(payload));
      const encrypted = await subtle.encrypt({ name: 'AES-GCM', iv }, key, plaintext);
      envelope = {
        format: 'rawnode-pm-state',
        version: 1,
        encrypted: true,
        iv: Buffer.from(iv).toString('base64'),
        data: Buffer.from(encrypted).toString('base64'),
      };

      envelope = { format: 'rawnode-pm-state', version: 1, encrypted: false, data: payload };
    }

    await atomicWriteFile(this.file, `${JSON.stringify(envelope, null, 2)}\n`, { encoding: 'utf8', mode: 0o600, dirMode: 0o700 });

  }

    let envelope;
    try { envelope = JSON.parse(await readFile(this.file, 'utf8')); }

      if (error?.code === 'ENOENT') return fallback;
      throw error;
    }

    if (envelope?.format !== 'rawnode-pm-state') throw new Error('Invalid RawNode PM state file.');
    if (!envelope.encrypted) return envelope.data;
    if (!this.keyProvider) throw new Error('State is encrypted but no key provider is configured.');
    const keyBytes = await this.keyProvider();
    const key = await importAesKey(keyBytes);
    const iv = Buffer.from(envelope.iv, 'base64');
    const ciphertext = Buffer.from(envelope.data, 'base64');
    const plaintext = await subtle.decrypt({ name: 'AES-GCM', iv }, key, ciphertext);
    return JSON.parse(decoder.decode(plaintext));
  }
}

import { open, readFile, rm } from 'node:fs/promises';
import path from 'node:path';
import { ensureDir } from '../support/fs.js';

function processAlive(pid) {
  if (!Number.isInteger(pid) || pid <= 0) return false;
  try { process.kill(pid, 0); return true; }
  catch (error) { return error?.code === 'EPERM'; }
}

export class DaemonLock {
  constructor(file) {
    this.file = file;
    this.handle = null;
  }

  async acquire() {
    await ensureDir(path.dirname(this.file), 0o700);
    for (let attempt = 0; attempt < 2; attempt++) {
      try {
        this.handle = await open(this.file, 'wx', 0o600);
        await this.handle.writeFile(`${process.pid}\n`);
        await this.handle.sync();
        return this;
      } catch (error) {
        if (error?.code !== 'EEXIST') throw error;
        let pid = 0;
        try { pid = Number((await readFile(this.file, 'utf8')).trim()); } catch {}
        if (processAlive(pid)) {
          const lockError = new Error(`RawNode PM daemon is already running with PID ${pid}.`);
          lockError.code = 'RNPM_DAEMON_ALREADY_RUNNING';
          throw lockError;
        }
        await rm(this.file, { force: true });
      }
    }
    throw new Error('Unable to acquire RawNode PM daemon lock.');
  }

  async release() {
    if (this.handle) {
      await this.handle.close().catch(() => {});
      this.handle = null;
    }
    await rm(this.file, { force: true }).catch(() => {});
  }
}

import { createWriteStream } from 'node:fs';
import { stat, rename, rm } from 'node:fs/promises';
import path from 'node:path';
import { ensureDir } from '../support/fs.js';
import { AsyncLock } from '../support/async.js';
import { safeJsonLine } from '../support/json.js';

export class AtomicJournal {
  constructor(options = {}) {
    this.file = options.file;
    this.maxBytes = Math.max(1024 * 1024, Number(options.maxBytes) || 20 * 1024 * 1024);
    this.maxFiles = Math.max(1, Number(options.maxFiles) || 3);
    this._stream = null;
    this._bytes = 0;
    this._lock = new AsyncLock();
  }

  async open() {
    if (this._stream) return this;
    if (!this.file) throw new Error('AtomicJournal requires file.');
    await ensureDir(path.dirname(this.file), 0o700);
    try { this._bytes = (await stat(this.file)).size; } catch { this._bytes = 0; }
    this._stream = createWriteStream(this.file, { flags: 'a', mode: 0o600 });
    this._stream.on('error', () => {});
    return this;
  }

  async append(type, payload = {}) {
    return this._lock.run(async () => {
      await this.open();
      const line = safeJsonLine({ at: Date.now(), type, payload });
      const bytes = Buffer.byteLength(line);
      if (this._bytes + bytes > this.maxBytes) await this._rotate();
      await new Promise((resolve, reject) => {
        if (this._stream.write(line)) return resolve();
        const onDrain = () => { cleanup(); resolve(); };
        const onError = (error) => { cleanup(); reject(error); };
        const cleanup = () => {
          this._stream.off('drain', onDrain);
          this._stream.off('error', onError);
        };
        this._stream.once('drain', onDrain);
        this._stream.once('error', onError);
      });
      this._bytes += bytes;
    });
  }

  async _rotate() {
    const stream = this._stream;
    this._stream = null;
    if (stream) await new Promise((resolve) => stream.end(resolve));
    await rm(`${this.file}.${this.maxFiles}`, { force: true }).catch(() => {});
    for (let i = this.maxFiles - 1; i >= 1; i--) {
      await rename(`${this.file}.${i}`, `${this.file}.${i + 1}`).catch((error) => {
        if (error?.code !== 'ENOENT') throw error;
      });
    }
    await rename(this.file, `${this.file}.1`).catch((error) => {
      if (error?.code !== 'ENOENT') throw error;
    });
    this._bytes = 0;
    await this.open();
  }

  async close() {
    if (!this._stream) return;
    const stream = this._stream;
    this._stream = null;
    await new Promise((resolve) => stream.end(resolve));
  }
}

export class ChangeCoalescer {
  constructor(callback, options = {}) {

    this.timer = null;

    this.changes.clear();
    await this.callback(values);
  }

  close() {
    if (this.timer) clearTimeout(this.timer);
    this.timer = null;
    this.changes.clear();
  }
}

async function walk(root, matcher, output, maxFiles, base = root) {
  const stack = [root];
  while (stack.length && output.size < maxFiles) {
    const current = stack.pop();
    let entries;
    try { entries = await readdir(current, { withFileTypes: true }); }
    catch { continue; }
    for (const entry of entries) {
      const absolute = path.join(current, entry.name);
      if (matcher.matches(absolute, base)) continue;
      if (entry.isDirectory()) stack.push(absolute);
      else if (entry.isFile()) {
        try {
          const info = await stat(absolute);
          output.set(absolute, `${info.mtimeMs}:${info.size}`);
        } catch {}
      }
      if (output.size >= maxFiles) break;
    }
  }
}

    await this._installNativeWatchers(session);
    session.pollTimer = setInterval(() => this._poll(session).catch((error) => this.emit('error', error, definition.name)), definition.watch.pollingIntervalMs);
    session.reconcileTimer = setInterval(() => this._installNativeWatchers(session).catch((error) => this.emit('error', error, definition.name)), definition.watch.reconcileIntervalMs);
    this.emit('watching', definition.name, roots);
    return session;
  }

  async _installNativeWatchers(session) {
    if (session.closed) return;
    const dirs = new Set();
    const stack = [...session.roots];
    while (stack.length && dirs.size < this.maxFiles) {
      const current = stack.pop();
      if (session.matcher.matches(current, session.definition.cwd)) continue;
      let entries;
      try { entries = await readdir(current, { withFileTypes: true }); }
      catch { continue; }
      dirs.add(current);
      for (const entry of entries) if (entry.isDirectory()) stack.push(path.join(current, entry.name));
    }

    for (const dir of dirs) {
      if (session.watchers.has(dir)) continue;
      try {
        const watcher = fsWatch(dir, { persistent: true }, (event, filename) => {
          const absolute = filename ? path.join(dir, String(filename)) : dir;
          if (!session.matcher.matches(absolute, session.definition.cwd)) session.coalescer.add(absolute, event);
          if (event === 'rename') queueMicrotask(() => this._installNativeWatchers(session).catch(() => {}));
        });
        watcher.on('error', (error) => this.emit('nativeError', error, session.name, dir));
        session.watchers.set(dir, watcher);
      } catch (error) {
        this.emit('nativeError', error, session.name, dir);
      }
    }

    for (const [dir, watcher] of session.watchers) {
      if (!dirs.has(dir)) {
        watcher.close();
        session.watchers.delete(dir);
      }
    }
  }

  async _buildSnapshot(session) {
    const next = new Map();
    for (const root of session.roots) await walk(root, session.matcher, next, this.maxFiles, session.definition.cwd);
    return next;
  }

  async _rebuildSnapshot(session) {
    session.snapshot = await this._buildSnapshot(session);
  }

  async _poll(session) {
    if (session.closed) return;
    const next = await this._buildSnapshot(session);
    for (const [file, signature] of next) {
      const previous = session.snapshot.get(file);
      if (previous == null) session.coalescer.add(file, 'created');
      else if (previous !== signature) session.coalescer.add(file, 'changed');
    }
    for (const file of session.snapshot.keys()) if (!next.has(file)) session.coalescer.add(file, 'deleted');
    session.snapshot = next;
  }

  async unwatch(name) {
    const session = this.sessions.get(name);
    if (!session) return false;
    session.closed = true;
    session.coalescer.close();
    if (session.pollTimer) clearInterval(session.pollTimer);
    if (session.reconcileTimer) clearInterval(session.reconcileTimer);
    for (const watcher of session.watchers.values()) watcher.close();
    session.watchers.clear();
    this.sessions.delete(name);
    this.emit('unwatched', name);
    return true;
  }

  status(name) {
    const session = this.sessions.get(name);
    if (!session) return null;
    return {
      enabled: true,
      mode: session.definition.watch.mode,
      roots: session.roots.slice(),
      nativeWatchers: session.watchers.size,
      trackedFiles: session.snapshot.size,
    };
  }

  async close() {
    await Promise.all(Array.from(this.sessions.keys(), (name) => this.unwatch(name)));
  }
}

export default WatchManager;

'use strict';

function metricName(value) {
  return String(value || 'rawnode_pm_metric').replace(/[^a-zA-Z0-9_:]/g, '_').replace(/^[^a-zA-Z_:]/, '_');
}

function escapeLabel(value) {
  return String(value ?? '').replace(/\\/g, '\\\\').replace(/"/g, '\\"').replace(/\n/g, '\\n');
}

function labels(value) {
  const entries = Object.entries(value || {}).filter(([, v]) => v != null);
  if (!entries.length) return '';
  return `{${entries.map(([key, item]) => `${metricName(key)}="${escapeLabel(item)}"`).join(',')}}`;
}

function line(out, name, value, labelSet = {}) {
  const number = Number(value);
  if (!Number.isFinite(number)) return;
  out.push(`${metricName(name)}${labels(labelSet)} ${number}`);
}

export function formatPrometheus({ registry, metrics }) {
  const out = [
    '# HELP rawnode_pm_up Whether the RawNode PM daemon is running.',
    '# TYPE rawnode_pm_up gauge',
    'rawnode_pm_up 1',
  ];
  const groups = registry?.list?.() || [];
  for (const group of groups) {
    const base = { app: group.name, namespace: group.namespace, mode: group.mode };
    line(out, 'rawnode_pm_desired_instances', group.desiredInstances, base);
    line(out, 'rawnode_pm_active_instances', group.activeInstances, base);
    line(out, 'rawnode_pm_quarantined', group.quarantined ? 1 : 0, base);
    for (const process of group.processes) {
      const labelSet = { ...base, slot: process.slot, pid: process.pid || 0, state: process.state };
      line(out, 'rawnode_pm_process_up', ['ready', 'running', 'online'].includes(process.state) ? 1 : 0, labelSet);
      line(out, 'rawnode_pm_process_uptime_milliseconds', process.uptimeMs, labelSet);
      const latest = process.metrics;
      if (!latest) continue;
      line(out, 'rawnode_pm_process_resident_memory_bytes', latest.memory?.rss, labelSet);
      line(out, 'rawnode_pm_process_heap_used_bytes', latest.memory?.heapUsed, labelSet);
      line(out, 'rawnode_pm_process_cpu_percent', latest.cpu?.percent, labelSet);
      line(out, 'rawnode_pm_process_event_loop_utilization_ratio', latest.eventLoop?.utilization, labelSet);
      line(out, 'rawnode_pm_process_event_loop_delay_p99_milliseconds', latest.eventLoop?.delayP99Ms, labelSet);
      line(out, 'rawnode_pm_process_active_requests', latest.adapter?.activeRequests, labelSet);
      line(out, 'rawnode_pm_process_open_sockets', latest.adapter?.openSockets, labelSet);
    }
  }
  const aggregate = metrics?.snapshot?.() || metrics || {};
  line(out, 'rawnode_pm_daemon_uptime_milliseconds', aggregate.uptimeMs || 0);
  for (const [name, value] of Object.entries(aggregate.counters || {})) line(out, `rawnode_pm_${name}`, value);
  return `${out.join('\n')}\n`;
}

export class PrometheusFormatter {
  format(input) { return formatPrometheus(input); }
}

import { EventEmitter } from 'node:events';

export class MetricsAggregator extends EventEmitter {
  constructor(options = {}) {
    super();
    this.historyLimit = Math.max(10, Number(options.historyLimit) || 300);
    this.processes = new Map();
    this.counters = Object.create(null);
    this.startedAt = Date.now();
  }

  inc(name, value = 1) {
    this.counters[name] = (this.counters[name] || 0) + Number(value || 0);
    return this.counters[name];
  }

  update(record, metrics) {
    if (!record || !metrics) return;
    let entry = this.processes.get(record.id);
    if (!entry) {
      entry = { name: record.name, slot: record.slot, pid: record.pid, latest: null, history: [] };
      this.processes.set(record.id, entry);
    }
    entry.pid = record.pid;
    entry.latest = metrics;
    entry.history.push(metrics);
    if (entry.history.length > this.historyLimit) entry.history.splice(0, entry.history.length - this.historyLimit);
    this.emit('metrics', record, metrics);
  }

  remove(record) {
    if (record) this.processes.delete(record.id);
  }

  snapshot({ includeHistory = false } = {}) {
    return {
      uptimeMs: Date.now() - this.startedAt,
      counters: { ...this.counters },
      processes: Array.from(this.processes.entries()).map(([id, entry]) => ({
        id,
        name: entry.name,
        slot: entry.slot,
        pid: entry.pid,
        latest: entry.latest,
        ...(includeHistory ? { history: entry.history.slice() } : {}),
      })),
    };
  }
}

export function controlRequest(command, params = {}, token = null) {
  return {
    protocol: CONTROL_PROTOCOL_VERSION,
    id: randomUUID(),
    token,
    command: String(command || ''),
    params: params && typeof params === 'object' ? params : {},
    at: Date.now(),
  };
}

export function controlResponse(request, result = null, error = null) {
  return {
    protocol: CONTROL_PROTOCOL_VERSION,
    id: request?.id || null,
    ok: !error,
    result: error ? null : result,
    error: error || null,
    at: Date.now(),
  };
}

export function controlEvent(event, payload = {}) {
  return {
    protocol: CONTROL_PROTOCOL_VERSION,
    stream: 'events',
    event: String(event || 'message'),
    payload,
    at: Date.now(),
  };
}

'use strict';

import http from 'node:http';
import { URL } from 'node:url';
import { serializeError } from '../errors.js';
import { safeStringify } from '../support/json.js';

function sendJson(res, status, value) {
  if (res.writableEnded || res.destroyed) return;
  const body = Buffer.from(safeStringify(value));
  res.statusCode = status;
  res.setHeader('content-type', 'application/json; charset=utf-8');
  res.setHeader('content-length', String(body.length));
  res.setHeader('cache-control', 'no-store');
  res.setHeader('x-content-type-options', 'nosniff');
  res.end(body);
}

async function readJson(req, maxBytes) {
  const declared = Number(req.headers['content-length']);
  if (Number.isFinite(declared) && declared > maxBytes) {
    const error = new Error('Management request body exceeds limit.');
    error.code = 'RNPM_BODY_TOO_LARGE';
    error.status = 413;
    throw error;
  }

  const chunks = [];
  let bytes = 0;
  for await (const chunk of req) {
    bytes += chunk.length;
    if (bytes > maxBytes) {
      const error = new Error('Management request body exceeds limit.');
      error.code = 'RNPM_BODY_TOO_LARGE';
      error.status = 413;
      throw error;
    }
    chunks.push(chunk);
  }
  if (!bytes) return {};
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch (cause) {
    const error = new Error('Management request body must contain valid JSON.', { cause });
    error.code = 'RNPM_BODY_INVALID_JSON';
    error.status = 400;
    throw error;
  }
}

function publicHealthView(health) {
  return {
    ok: health.ok,
    ready: health.ready,
    uptimeMs: health.uptimeMs,
    groups: health.groups,
    activeProcesses: health.activeProcesses,
  };
}

export class ManagementHttpServer {
  constructor(options = {}) {
    this.host = options.host || '127.0.0.1';
    this.port = options.port == null ? null : Number(options.port);
    this._enabled = options.enabled ?? options.port != null;
    this.authenticator = options.authenticator;
    this.manager = options.manager;
    this.publicHealth = options.publicHealth !== false;
    this.publicMetrics = options.publicMetrics === true;
    this.maxBodyBytes = Math.max(4_096, Number(options.maxBodyBytes) || 1024 * 1024);
    this.maxSseBufferBytes = Math.max(64 * 1024, Number(options.maxSseBufferBytes) || 4 * 1024 * 1024);
    this.server = null;
    this.sse = new Set();
    this._eventHandler = (packet) => this._broadcast(packet);
  }

  enabled() {
    return this._enabled === true && Number.isInteger(this.port) && this.port >= 0 && this.port <= 65535;
  }

  authorized(req) {
    const header = String(req.headers.authorization || '');
    const token = header.startsWith('Bearer ') ? header.slice(7).trim() : '';
    return this.authenticator?.verify?.(token) === true;
  }

  async start() {
    if (!this.enabled() || this.server) return this;
    this.server = http.createServer((req, res) => this._handle(req, res).catch((error) => {
      sendJson(res, error?.status || 500, { ok: false, error: serializeError(error) });
    }));
    this.server.requestTimeout = 15_000;
    this.server.headersTimeout = 10_000;
    this.server.keepAliveTimeout = 5_000;
    this.server.maxHeadersCount = 100;
    this.server.on('clientError', (_error, socket) => {
      if (!socket.writable) return socket.destroy();
      socket.end('HTTP/1.1 400 Bad Request\r\nConnection: close\r\n\r\n');
    });
    try {
      await new Promise((resolve, reject) => {
        const onError = (error) => { cleanup(); reject(error); };
        const onListening = () => { cleanup(); resolve(); };
        const cleanup = () => {
          this.server.off('error', onError);
          this.server.off('listening', onListening);
        };
        this.server.once('error', onError);
        this.server.once('listening', onListening);
        this.server.listen(this.port, this.host);
      });
    } catch (error) {
      this.server?.close?.();
      this.server = null;
      throw error;
    }
    this.manager?.realtime?.on?.('event', this._eventHandler);
    return this;
  }

  async _handle(req, res) {
    const url = new URL(req.url || '/', `http://${req.headers.host || `${this.host}:${this.port}`}`);
    const method = String(req.method || 'GET').toUpperCase();
    const authenticated = this.authorized(req);

    if (method === 'GET' && url.pathname === '/health') {
      if (!this.publicHealth && !authenticated) return sendJson(res, 401, { error: 'management token required' });
      const health = this.manager.health();
      return sendJson(res, health.ok ? 200 : 503, authenticated ? health : publicHealthView(health));
    }
    if (method === 'GET' && url.pathname === '/ready') {
      if (!this.publicHealth && !authenticated) return sendJson(res, 401, { error: 'management token required' });
      const health = this.manager.health();
      return sendJson(res, health.ready ? 200 : 503, authenticated ? health : publicHealthView(health));
    }
    if (method === 'GET' && url.pathname === '/metrics') {
      if (!this.publicMetrics && !authenticated) return sendJson(res, 401, { error: 'management token required' });
      const result = await this.manager.handleCommand('prometheus', {}, { requestId: null, transport: 'http' });
      res.statusCode = 200;
      res.setHeader('content-type', result.contentType);
      res.setHeader('cache-control', 'no-store');
      res.setHeader('x-content-type-options', 'nosniff');
      res.end(result.body);
      return;
    }

    if (!authenticated) return sendJson(res, 401, { error: 'management token required' });

    if (method === 'GET' && url.pathname === '/v1/status') {
      const name = url.searchParams.get('name');
      return sendJson(res, 200, await this.manager.handleCommand('status', { name }, { transport: 'http' }));
    }
    if (method === 'GET' && url.pathname === '/v1/events') {
      return this._openSse(req, res, url);
    }
    if (method === 'POST' && url.pathname === '/v1/command') {
      const body = await readJson(req, this.maxBodyBytes);
      if (!body.command) {
        const error = new Error('command is required.');
        error.status = 400;
        throw error;
      }
      const result = await this.manager.handleCommand(body.command, body.params || {}, { requestId: body.id || null, transport: 'http' });
      return sendJson(res, 200, { ok: true, result });
    }

    return sendJson(res, 404, { error: 'not found' });
  }

  _openSse(req, res, url) {
    const client = {
      res,
      app: url.searchParams.get('app'),
      events: new Set((url.searchParams.get('events') || '*').split(',').map((value) => value.trim()).filter(Boolean)),
    };
    res.statusCode = 200;
    res.setHeader('content-type', 'text/event-stream; charset=utf-8');
    res.setHeader('cache-control', 'no-cache, no-transform');
    res.setHeader('connection', 'keep-alive');
    res.setHeader('x-content-type-options', 'nosniff');
    res.flushHeaders?.();
    res.write(`event: ready\ndata: ${safeStringify({ pid: process.pid, at: Date.now() })}\n\n`);
    this.sse.add(client);
    const close = () => this.sse.delete(client);
    req.once('close', close);
    res.once('close', close);
  }

  _broadcast(packet) {
    for (const client of this.sse) {
      if (client.res.destroyed || client.res.writableEnded) {
        this.sse.delete(client);
        continue;
      }
      const acceptsEvent = client.events.has('*') || client.events.has(packet.event);
      const name = packet.payload?.name || packet.payload?.record?.name || null;
      const acceptsApp = !client.app || client.app === name;
      if (!acceptsEvent || !acceptsApp) continue;
      if (client.res.writableLength > this.maxSseBufferBytes) {
        client.res.destroy(new Error('Slow SSE client exceeded buffer limit.'));
        this.sse.delete(client);
        continue;
      }
      client.res.write(`event: ${packet.event}\ndata: ${safeStringify(packet)}\n\n`);
    }
  }

  address() {
    try { return this.server?.address?.() || null; } catch { return null; }
  }

  async close() {
    this.manager?.realtime?.off?.('event', this._eventHandler);
    for (const client of this.sse) client.res.destroy();
    this.sse.clear();
    if (!this.server) return;
    const server = this.server;
    this.server = null;
    await new Promise((resolve) => server.close(resolve));
  }
}

'use strict';

import { EventEmitter } from 'node:events';
import net from 'node:net';
import { chmod } from 'node:fs/promises';
import { CONTROL_PROTOCOL_VERSION, DEFAULTS } from '../constants.js';
import { controlResponse, controlEvent } from './ControlProtocol.js';
import { serializeError } from '../errors.js';
import { removeIfExists } from '../support/fs.js';
import { safeJsonLine } from '../support/json.js';

export class ControlServer extends EventEmitter {
  constructor(options = {}) {
    super();
    this.socketPath = options.socketPath;
    this.authenticator = options.authenticator;
    this.handler = options.handler;
    this.maxFrameBytes = Math.max(4_096, Number(options.maxFrameBytes) || DEFAULTS.controlMaxFrameBytes);
    this.maxSubscriberBytes = Math.max(64 * 1024, Number(options.maxSubscriberBytes) || 4 * 1024 * 1024);
    this.server = null;
    this._serverErrorHandler = (error) => {
      if (this.listenerCount('error') > 0) this.emit('error', error);
      else this.emit('serverError', error);
    };
    this.connections = new Set();
    this.subscribers = new Map();
  }

  async start() {
    if (this.server) return this;
    if (!this.socketPath) throw new Error('ControlServer requires socketPath.');
    if (process.platform !== 'win32') await removeIfExists(this.socketPath);
    this.server = net.createServer((socket) => this._onConnection(socket));
    try {
      await new Promise((resolve, reject) => {
        const onError = (error) => { cleanup(); reject(error); };
        const onListening = () => { cleanup(); resolve(); };
        const cleanup = () => {
          this.server.off('error', onError);
          this.server.off('listening', onListening);
        };
        this.server.once('error', onError);
        this.server.once('listening', onListening);
        this.server.listen(this.socketPath);
      });
    } catch (error) {
      this.server?.close?.();
      this.server = null;
      if (process.platform !== 'win32') await removeIfExists(this.socketPath);
      throw error;
    }
    this.server.on('error', this._serverErrorHandler);
    if (process.platform !== 'win32') {
      try { await chmod(this.socketPath, 0o600); } catch {}
    }
    this.emit('listening', this.socketPath);
    return this;
  }

  _onConnection(socket) {
    socket.setNoDelay(true);
    socket.setKeepAlive(true, 30_000);
    socket._rnBuffer = '';
    socket._rnAuthorized = false;
    this.connections.add(socket);
    socket.on('data', (chunk) => this._onData(socket, chunk));
    socket.on('error', (error) => this.emit('clientError', error));
    socket.on('close', () => {
      this.connections.delete(socket);
      this.subscribers.delete(socket);
    });
  }

  _onData(socket, chunk) {
    socket._rnBuffer += chunk.toString('utf8');
    if (Buffer.byteLength(socket._rnBuffer) > this.maxFrameBytes) {
      this._write(socket, controlResponse(null, null, { code: 'RNPM_FRAME_TOO_LARGE', message: 'Control frame exceeds limit.' }));
      socket.destroy();
      return;
    }
    let index;
    while ((index = socket._rnBuffer.indexOf('\n')) !== -1) {
      const line = socket._rnBuffer.slice(0, index);
      socket._rnBuffer = socket._rnBuffer.slice(index + 1);
      if (!line.trim()) continue;
      this._handleLine(socket, line).catch((error) => {
        this._write(socket, controlResponse(null, null, serializeError(error)));
      });
    }
  }

  async _handleLine(socket, line) {
    let request;
    try { request = JSON.parse(line); }
    catch {
      this._write(socket, controlResponse(null, null, { code: 'RNPM_BAD_JSON', message: 'Invalid JSON control frame.' }));
      return;
    }
    if (request.protocol !== CONTROL_PROTOCOL_VERSION) {
      this._write(socket, controlResponse(request, null, { code: 'RNPM_PROTOCOL_MISMATCH', message: `Expected protocol ${CONTROL_PROTOCOL_VERSION}.` }));
      return;
    }
    if (!this.authenticator?.verify?.(request.token)) {
      this._write(socket, controlResponse(request, null, { code: 'RNPM_UNAUTHORIZED', message: 'Invalid control token.', status: 401 }));
      socket.end();
      return;
    }
    socket._rnAuthorized = true;

    if (request.command === 'subscribe') {
      this.subscribers.set(socket, {
        events: new Set(Array.isArray(request.params?.events) ? request.params.events.map(String) : ['*']),
        app: request.params?.app || null,
      });
      this._write(socket, controlResponse(request, { subscribed: true }));
      return;
    }

    try {
      const result = await this.handler(request.command, request.params || {}, {
        requestId: request.id,
        socket,
      });
      this._write(socket, controlResponse(request, result));
    } catch (error) {
      this._write(socket, controlResponse(request, null, serializeError(error, { stack: false })));
    }
  }

  _write(socket, value) {
    if (!socket.writable || socket.destroyed) return false;
    if (socket.writableLength > this.maxSubscriberBytes) {
      socket.destroy(new Error('Slow control client exceeded buffer limit.'));
      return false;
    }
    return socket.write(safeJsonLine(value));
  }

  broadcast(event, payload = {}) {
    const packet = controlEvent(event, payload);
    for (const [socket, filter] of this.subscribers) {
      if (socket.destroyed) { this.subscribers.delete(socket); continue; }
      const acceptsEvent = filter.events.has('*') || filter.events.has(event);
      const acceptsApp = !filter.app || payload?.name === filter.app || payload?.record?.name === filter.app;
      if (acceptsEvent && acceptsApp) this._write(socket, packet);
    }
    this.emit('broadcast', event, payload);
  }

  async close() {
    for (const socket of this.connections) socket.destroy();
    this.connections.clear();
    this.subscribers.clear();
    if (this.server) {
      const server = this.server;
      this.server = null;
      await new Promise((resolve) => server.close(resolve));
    }
    if (process.platform !== 'win32') await removeIfExists(this.socketPath);
  }
}

export default ControlServer;

import net from 'node:net';
import { readFile } from 'node:fs/promises';
import { controlRequest } from './ControlProtocol.js';
import { withTimeout } from '../support/async.js';
import { ControlError } from '../errors.js';

export class ControlClient {
  constructor(options = {}) {
    this.socketPath = options.socketPath;
    this.tokenPath = options.tokenPath;
    this.token = options.token || null;
    this.timeoutMs = Math.max(250, Number(options.timeoutMs) || 10_000);
    this.maxFrameBytes = Math.max(4_096, Number(options.maxFrameBytes) || 1024 * 1024);
  }

  async initialize() {
    if (!this.token) this.token = (await readFile(this.tokenPath, 'utf8')).trim();
    return this;
  }

  async request(command, params = {}, options = {}) {
    await this.initialize();
    const request = controlRequest(command, params, this.token);
    let socket = null;
    const task = new Promise((resolve, reject) => {
      socket = net.connect(this.socketPath);
      let buffer = '';
      const cleanup = () => socket?.destroy();
      socket.setNoDelay(true);
      socket.once('connect', () => socket.write(`${JSON.stringify(request)}\n`));
      socket.on('data', (chunk) => {
        buffer += chunk.toString('utf8');
        if (Buffer.byteLength(buffer) > this.maxFrameBytes) {
          cleanup();
          reject(new ControlError('Control response exceeds frame limit.', { code: 'RNPM_FRAME_TOO_LARGE' }));
          return;
        }
        const index = buffer.indexOf('\n');
        if (index === -1) return;
        const line = buffer.slice(0, index);
        cleanup();
        try {
          const response = JSON.parse(line);
          if (!response.ok) {
            reject(new ControlError(response.error?.message || 'Control request failed.', {
              code: response.error?.code || 'RNPM_CONTROL_REQUEST_FAILED',
              status: response.error?.status || 500,
              details: response.error?.details || null,
            }));
          } else resolve(response.result);
        } catch (error) { reject(error); }
      });
      socket.once('error', reject);
    });
    try {
      return await withTimeout(task, options.timeoutMs || this.timeoutMs, { message: `Control command '${command}' timed out.` });
    } finally {
      socket?.destroy();
    }
  }

  async subscribe(options = {}, onEvent = () => {}) {
    await this.initialize();
    const request = controlRequest('subscribe', options, this.token);
    const socket = net.connect(this.socketPath);
    let buffer = '';
    let acknowledged = false;
    const ready = new Promise((resolve, reject) => {
      socket.once('connect', () => socket.write(`${JSON.stringify(request)}\n`));
      socket.once('error', reject);
      socket.on('data', (chunk) => {
        buffer += chunk.toString('utf8');
        if (Buffer.byteLength(buffer) > this.maxFrameBytes) {
          socket.destroy();
          reject(new ControlError('Control event frame exceeds limit.', { code: 'RNPM_FRAME_TOO_LARGE' }));
          return;
        }
        let index;
        while ((index = buffer.indexOf('\n')) !== -1) {
          const line = buffer.slice(0, index);
          buffer = buffer.slice(index + 1);
          if (!line.trim()) continue;
          let packet;
          try { packet = JSON.parse(line); } catch { continue; }
          if (!acknowledged && packet.id === request.id) {
            if (!packet.ok) return reject(new ControlError(packet.error?.message || 'Subscription failed.', { code: packet.error?.code }));
            acknowledged = true;
            resolve();
            continue;
          }
          if (packet.stream === 'events') {
            try { onEvent(packet); }
            catch (error) {
              if (typeof options.onError === 'function') options.onError(error, packet);
              else socket.destroy(error);
            }
          }
        }
      });
    });
    try {
      await withTimeout(ready, this.timeoutMs, { message: 'Control subscription timed out.' });
      return () => socket.destroy();
    } catch (error) {
      socket.destroy();
      throw error;
    }
  }
}

export default ControlClient;

'use strict';

export function parseArgs(argv = []) {
  const positionals = [];
  const options = Object.create(null);
  for (let index = 0; index < argv.length; index++) {
    const value = argv[index];
    if (value === '--') {
      positionals.push(...argv.slice(index + 1));
      break;
    }
    if (value.startsWith('--no-')) {
      options[value.slice(5)] = false;
      continue;
    }
    if (value.startsWith('--')) {
      const raw = value.slice(2);
      const equal = raw.indexOf('=');
      if (equal !== -1) {
        options[raw.slice(0, equal)] = raw.slice(equal + 1);
        continue;
      }
      const next = argv[index + 1];
      if (next != null && !next.startsWith('-')) {
        options[raw] = next;
        index++;
      } else options[raw] = true;
      continue;
    }
    if (value.startsWith('-') && value.length > 1) {
      for (const flag of value.slice(1)) options[flag] = true;
      continue;
    }
    positionals.push(value);
  }
  return { command: positionals.shift() || 'status', positionals, options };
}

export function numberOption(value, fallback = undefined) {
  const number = Number(value);
  return Number.isFinite(number) ? number : fallback;
}

'use strict';

import { formatDuration } from '../support/duration.js';

const colorEnabled = () => !!(process.stdout.isTTY && !process.env.NO_COLOR);
const wrap = (code, value) => colorEnabled() ? `\x1b[${code}m${value}\x1b[0m` : String(value);

export const style = Object.freeze({
  green: (value) => wrap('32', value),
  yellow: (value) => wrap('33', value),
  red: (value) => wrap('31', value),
  cyan: (value) => wrap('36', value),
  gray: (value) => wrap('90', value),
  bold: (value) => wrap('1', value),
});

export function formatBytes(value) {
  const bytes = Number(value) || 0;
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 ** 2) return `${(bytes / 1024).toFixed(1)} KiB`;
  if (bytes < 1024 ** 3) return `${(bytes / 1024 ** 2).toFixed(1)} MiB`;
  return `${(bytes / 1024 ** 3).toFixed(2)} GiB`;
}

function stateText(state) {
  if (['running', 'ready', 'online'].includes(state)) return style.green(state);
  if (['starting', 'draining', 'stopping', 'backoff'].includes(state)) return style.yellow(state);
  if (['crashed', 'unhealthy', 'quarantined', 'boot_failed'].includes(state)) return style.red(state);
  return style.gray(state || 'unknown');
}

export function renderTable(rows) {
  if (!rows.length) return '';
  const widths = [];
  for (const row of rows) row.forEach((cell, index) => { widths[index] = Math.max(widths[index] || 0, String(cell.raw ?? cell).length); });
  return rows.map((row, rowIndex) => {
    const line = row.map((cell, index) => {
      const raw = String(cell.raw ?? cell);
      const shown = String(cell.shown ?? cell);
      return shown + ' '.repeat(Math.max(0, widths[index] - raw.length));
    }).join('  ').trimEnd();
    if (rowIndex === 0) return `${style.bold(line)}\n${widths.map((width) => '─'.repeat(width)).join('  ')}`;
    return line;
  }).join('\n');
}

export function renderStatus(groups = []) {
  const rows = [['APP', 'SLOT', 'PID', 'MODE', 'STATE', 'UPTIME', 'CPU', 'MEMORY', 'RESTARTS']];
  for (const group of groups) {
    if (!group.processes.length) {
      rows.push([group.name, '-', '-', group.mode, { raw: group.quarantined ? 'quarantined' : 'stopped', shown: stateText(group.quarantined ? 'quarantined' : 'stopped') }, '-', '-', '-', group.restart?.restartTimes?.length || 0]);
      continue;
    }
    const active = new Set();
    for (const process of group.processes) {
      const key = process.slot;
      if (active.has(key) && ['stopped', 'crashed'].includes(process.state)) continue;
      active.add(key);
      rows.push([
        group.name,
        process.slot,
        process.pid || '-',
        group.mode,
        { raw: process.state, shown: stateText(process.state) },
        process.uptimeMs ? formatDuration(process.uptimeMs) : '-',
        process.metrics?.cpu?.percent != null ? `${process.metrics.cpu.percent.toFixed(1)}%` : '-',
        process.metrics?.memory?.rss != null ? formatBytes(process.metrics.memory.rss) : '-',
        group.restart?.restartTimes?.length || 0,
      ]);
    }
  }
  return renderTable(rows);
}

export function print(value, options = {}) {
  if (options.json) process.stdout.write(`${JSON.stringify(value, null, 2)}\n`);
  else if (typeof value === 'string') process.stdout.write(value.endsWith('\n') ? value : `${value}\n`);
  else process.stdout.write(`${JSON.stringify(value, null, 2)}\n`);
}

'use strict';

import { ProcessManager } from '../foundation/ProcessManager.js';
import { parseArgs, numberOption } from './arguments.js';
import { ensureDaemon, createClient, pingDaemon, RAWNODE_PM_BIN } from './daemon.js';
import { renderStatus, print, style } from './format.js';
import { StartupGenerator } from '../integrations/StartupGenerator.js';
import { RawNodeIntegrator } from '../integrations/RawNodeIntegrator.js';
import { atomicWriteFile } from '../support/fs.js';
import { resolveStatePaths } from '../support/paths.js';

function usage() {
  return `RawNode PM — RawNode-aware runtime control plane

Usage:
  rawnode-pm start [manifest] [--name app] [--instances N] [--force]
  rawnode-pm stop <app>
  rawnode-pm restart <app>
  rawnode-pm reload <app>
  rawnode-pm scale <app> <instances>
  rawnode-pm status [app] [--json]
  rawnode-pm show <app> [--json]
  rawnode-pm logs <app> [--lines 100] [--stream stdout|stderr] [--follow]
  rawnode-pm monitor [--app name]
  rawnode-pm metrics [--prometheus] [--json]
  rawnode-pm health [--json]
  rawnode-pm doctor [--json]
  rawnode-pm diagnostics <app>
  rawnode-pm save
  rawnode-pm resurrect
  rawnode-pm startup [systemd|launchd] [--output file] [--user-service]

Global options:

  --cwd <path>         Project directory

async function integrateRawNode(options, global) {
  const integrator = new RawNodeIntegrator({
    root: options.root || global.cwd,
    applicationFile: options.application,
    serveFile: options.serve,
    configFile: options.config,
    manifestFile: options.manifest,
  });
  const result = await integrator.integrate({
    dryRun: options['dry-run'] === true || options.dryRun === true,
    force: options.force === true,
  });
  print(result, { json: true });
  return 0;
}

async function generateStartup(positionals, options, global) {
  const platform = positionals[0] || options.platform || process.platform;
  const stateDir = resolveStatePaths(global.stateDir).root;
  const generator = new StartupGenerator({
    nodePath: process.execPath,
    binPath: options.bin || RAWNODE_PM_BIN,
    stateDir,
    cwd: global.cwd,
    user: options.user || null,
    group: options.group || null,
    userService: options['user-service'] === true,
    label: options.label || 'org.rawnode.pm',
    resurrect: options.resurrect !== false,
  });
  const result = generator.generate(platform);
  if (options.output) {
    const file = path.resolve(global.cwd, String(options.output));
    await atomicWriteFile(file, result.content, { encoding: 'utf8', mode: 0o600 });
    print({ ok: true, platform: result.platform, file, installTarget: result.installTarget }, { json: options.json });
  } else {
    process.stdout.write(result.content);
  }
  return 0;
}

async function runDaemon(options) {
  const manager = new ProcessManager({ stateDir: options['state-dir'], cwd: options.cwd });
  const done = new Promise((resolve) => manager.once('closed', resolve));
  const shutdown = () => manager.close().catch((error) => {
    process.stderr.write(`${error?.stack || error}\n`);
    process.exitCode = 1;
  });
  process.once('SIGTERM', shutdown);
  process.once('SIGINT', shutdown);
  process.once('SIGHUP', () => manager.saveState().catch(() => {}));
  await manager.start({ resurrect: options.resurrect === true });
  await done;
  return 0;
}

async function followLogs(client, name, options) {
  const initial = await client.request('logs', { name, lines: numberOption(options.lines, 100), stream: options.stream });
  for (const entry of initial) process.stdout.write(`[${new Date(entry.at).toISOString()}] ${entry.name}:${entry.slot} ${entry.stream} ${entry.line}\n`);
  const close = await client.subscribe({ events: ['log:line'], app: name }, (packet) => {
    const entry = packet.payload;
    if (options.stream && entry.stream !== options.stream) return;
    process.stdout.write(`[${new Date(entry.at).toISOString()}] ${entry.name}:${entry.slot} ${entry.stream} ${entry.line}\n`);
  });
  await new Promise((resolve) => {
    const stop = () => { close(); resolve(); };
    process.once('SIGINT', stop);
    process.once('SIGTERM', stop);
  });
}

async function monitor(client, options) {
  let stopped = false;
  let rendering = false;
  let dirty = true;
  const close = await client.subscribe({ events: ['*'], app: options.app || null }, () => { dirty = true; });
  const render = async () => {
    if (rendering || stopped || !dirty) return;
    rendering = true;
    dirty = false;
    try {
      const groups = await client.request('status', { name: options.app || null });
      const values = Array.isArray(groups) ? groups : [groups];
      if (process.stdout.isTTY) process.stdout.write('\x1b[2J\x1b[H');
      process.stdout.write(`${style.bold('RawNode PM Monitor')}  ${new Date().toISOString()}\n\n${renderStatus(values)}\n`);
    } catch (error) {
      process.stderr.write(`${error.message}\n`);
    } finally { rendering = false; }
  };
  const timer = setInterval(() => render(), 1_000);
  await render();
  await new Promise((resolve) => {
    const stop = () => { stopped = true; clearInterval(timer); close(); resolve(); };
    process.once('SIGINT', stop);
    process.once('SIGTERM', stop);
  });
}

export async function runCli(argv = process.argv.slice(2)) {
  const { command, positionals, options } = parseArgs(argv);
  const global = { stateDir: options['state-dir'], cwd: options.cwd || process.cwd() };
  if (command === 'help' || options.help || options.h) {
    process.stdout.write(usage());
    return 0;
  }
  if (command === 'daemon') return runDaemon(options);
  if (command === 'startup') return generateStartup(positionals, options, global);
  if (command === 'integrate') return integrateRawNode(options, global);

  if (command === 'kill') {
    const running = await pingDaemon(global);
    if (!running) {
      print({ ok: true, running: false }, { json: options.json });
      return 0;
    }
    const result = await createClient(global).request('shutdown-daemon');
    print(result, { json: options.json });
    return 0;
  }

  const client = await ensureDaemon(global);
  let result;
  switch (command) {
    case 'start':
      result = await client.request('start', {
        manifest: positionals[0] || options.manifest || null,
        name: options.name || null,
        all: options.all === true,
        force: options.force === true,
        instances: numberOption(options.instances),
        cwd: global.cwd,
      }, { timeoutMs: numberOption(options.timeout, 120_000) });
      break;
    case 'stop': result = await client.request('stop', { name: positionals[0] || options.name }); break;
    case 'restart': result = await client.request('restart', { name: positionals[0] || options.name, force: options.force === true }, { timeoutMs: 120_000 }); break;
    case 'reload': result = await client.request('reload', { name: positionals[0] || options.name }, { timeoutMs: 120_000 }); break;
    case 'scale': result = await client.request('scale', { name: positionals[0] || options.name, instances: numberOption(positionals[1] || options.instances) }, { timeoutMs: 120_000 }); break;
    case 'delete': result = await client.request('delete', { name: positionals[0] || options.name }); break;
    case 'show': result = await client.request('show', { name: positionals[0] || options.name }); break;
    case 'status': {
      result = await client.request('status', { name: positionals[0] || options.name || null });
      if (!options.json) {
        const groups = Array.isArray(result) ? result : [result];
        print(renderStatus(groups));
        return 0;
      }
      break;
    }
    case 'logs': {
      const name = positionals[0] || options.name;
      if (options.follow || options.f) {
        await followLogs(client, name, options);
        return 0;
      }
      result = await client.request('logs', { name, lines: numberOption(options.lines, 100), stream: options.stream });
      if (!options.json) {
        for (const entry of result) process.stdout.write(`[${new Date(entry.at).toISOString()}] ${entry.name}:${entry.slot} ${entry.stream} ${entry.line}\n`);
        return 0;
      }
      break;
    }
    case 'monitor':
      await monitor(client, options);
      return 0;
    case 'metrics':
      if (options.prometheus) {
        const response = await client.request('prometheus');
        process.stdout.write(response.body);
        return 0;
      }
      result = await client.request('metrics', { history: options.history === true });
      break;
    case 'health': result = await client.request('health'); break;
    case 'doctor': result = await client.request('doctor'); break;
    case 'diagnostics': result = await client.request('diagnostics', { name: positionals[0] || options.name, slot: numberOption(options.slot), logLines: numberOption(options.lines) }, { timeoutMs: 120_000 }); break;
    case 'save': result = await client.request('save'); break;
    case 'resurrect': result = await client.request('resurrect', { force: options.force === true }, { timeoutMs: 120_000 }); break;
    case 'history': result = await client.request('history', { event: options.event, limit: numberOption(options.limit, 100) }); break;
    default:
      process.stderr.write(`Unknown command '${command}'.\n\n${usage()}`);
      return 2;
  }
  print(result, { json: options.json || typeof result !== 'string' });
  return 0;
}

export default runCli;

'use strict';

import { spawn } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { ControlClient } from '../control/ControlClient.js';
import { resolveStatePaths } from '../support/paths.js';
import { sleep } from '../support/async.js';

export const RAWNODE_PM_BIN = fileURLToPath(new URL('../../bin/rawnode-pm.js', import.meta.url));

export function createClient(options = {}) {
  const paths = resolveStatePaths(options.stateDir);
  return new ControlClient({ socketPath: paths.socket, tokenPath: paths.token, timeoutMs: options.timeoutMs || 10_000 });
}

export async function pingDaemon(options = {}) {
  try { return await createClient(options).request('ping', {}, { timeoutMs: options.timeoutMs || 750 }); }
  catch { return null; }
}

export async function ensureDaemon(options = {}) {
  const existing = await pingDaemon(options);
  if (existing) return createClient(options);

  const args = [RAWNODE_PM_BIN, 'daemon'];
  if (options.stateDir) args.push('--state-dir', options.stateDir);
  if (options.cwd) args.push('--cwd', options.cwd);
  if (options.resurrect) args.push('--resurrect');
  const child = spawn(process.execPath, args, {
    detached: true,
    stdio: 'ignore',
    cwd: options.cwd || process.cwd(),
    env: { ...process.env },
    windowsHide: true,
  });
  child.unref();

  const deadline = Date.now() + (options.startTimeoutMs || 10_000);
  let lastError = null;
  while (Date.now() < deadline) {
    try {
      const client = createClient(options);
      await client.request('ping', {}, { timeoutMs: 500 });
      return client;
    } catch (error) { lastError = error; }
    await sleep(100);
  }
  const error = new Error(`RawNode PM daemon did not become ready.${lastError ? ` ${lastError.message}` : ''}`);
  error.code = 'RNPM_DAEMON_START_TIMEOUT';
  throw error;
}
